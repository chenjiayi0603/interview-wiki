### 5.1.1 降低时间复杂度

```cpp
// ❌ O(n²) 算法
bool has_duplicate_naive(const std::vector<int>& arr) {
    for (size_t i = 0; i < arr.size(); ++i) {
        for (size_t j = i + 1; j < arr.size(); ++j) {
            if (arr[i] == arr[j]) return true;
        }
    }
    return false;
}

// ✅ O(n) 算法
bool has_duplicate(const std::vector<int>& arr) {
    std::unordered_set<int> seen;
    for (int val : arr) {
        if (seen.count(val)) return true;
        seen.insert(val);
    }
    return false;
}
```

## 5.2 数据结构优化

### 5.2.1 小对象优化（SOO）

```cpp
// std::string 的小字符串优化
std::string small = "hello";  // 可能存储在对象内部，不分配堆内存
std::string large(1000, 'a'); // 分配堆内存

// 自定义小对象优化
template<size_t N>
class SmallString {
private:
    size_t size_;
    union {
        char small_[N];      // 小字符串存储在栈上
        char* large_;        // 大字符串存储在堆上
    };
    
public:
    SmallString(const char* str) {
        size_t len = strlen(str);
        if (len < N) {
            size_ = len;
            memcpy(small_, str, len + 1);
        } else {
            size_ = len;
            large_ = new char[len + 1];
            memcpy(large_, str, len + 1);
        }
    }
};
```

## 5.3 查找优化

### 5.3.1 二分查找优化

```cpp
// 标准二分查找
int binary_search(const std::vector<int>& arr, int target) {
    int left = 0, right = arr.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) return mid;
        if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// ✅ 优化：减少分支
int binary_search_optimized(const std::vector<int>& arr, int target) {
    size_t left = 0;
    size_t len = arr.size();
    while (len > 0) {
        size_t half = len / 2;
        size_t mid = left + half;
        if (arr[mid] < target) {
            left = mid + 1;
            len = len - half - 1;
        } else {
            len = half;
        }
    }
    return (left < arr.size() && arr[left] == target) ? left : -1;
}
```

// 自定义哈希函数（如果标准哈希不够好）
struct CustomHash {
    size_t operator()(const std::string& key) const {
        // 使用更好的哈希算法（如 FNV-1a）

        return hash;
    }
