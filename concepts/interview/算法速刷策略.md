```cpp
// 返回下标，无序数组、一次哈希 O(n)
vector<int> twoSum(vector<int>& a, int t) {
  unordered_map<int, int> m;
  for (int i = 0; i < (int)a.size(); ++i) {
    int c = t - a[i];
    if (m.count(c)) return {m[c], i};
    m[a[i]] = i;
  }
  return {};
}
```

#### 数组 · 三数之和

```cpp
// 排序+双指针，去重
vector<vector<int>> threeSum(vector<int>& a) {
  int n = a.size();
  sort(a.begin(), a.end());
  vector<vector<int>> r;
  for (int i = 0; i < n; ++i) {
    if (i && a[i] == a[i - 1]) continue;
    int L = i + 1, R = n - 1;
    while (L < R) {
      int s = a[i] + a[L] + a[R];
      if (s < 0) ++L;
      else if (s > 0) --R;
      else {
        r.push_back({a[i], a[L], a[R]});
        int lv = a[L], rv = a[R];
        while (L < R && a[L] == lv) ++L;
        while (L < R && a[R] == rv) --R;
      }
    }
  }
  return r;
}
```

// Kadane：以 i 结尾的最大和
int maxSubArray(vector<int>& a) {
  int me = a[0], g = a[0];
  for (int i = 1; i < (int)a.size(); ++i) {
    me = max(a[i], me + a[i]);
    g = max(g, me);
  }
  return g;
}

ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
  ListNode d, *t = &d;
  while (l1 && l2) {
    if (l1->val <= l2->val) { t->next = l1; l1 = l1->next; }
    else { t->next = l2; l2 = l2->next; }
    t = t->next;
  }
  t->next = l1 ? l1 : l2;
  return d.next;
}

bool hasCycle(ListNode* h) {
  if (!h || !h->next) return false;
  ListNode *s = h, *f = h->next;
  while (f && f->next) {
    if (s == f) return true;
    s = s->next;
    f = f->next->next;
  }
  return false;
}

vector<vector<int>> levelOrder(TreeNode* r) {
  vector<vector<int>> a;
  if (!r) return a;
  queue<TreeNode*> q;
  q.push(r);
  while (!q.empty()) {
    int n = q.size();
    vector<int> row;
    for (int i = 0; i < n; ++i) {
      TreeNode* t = q.front();
      q.pop();
      row.push_back(t->val);
      if (t->left) q.push(t->left);
      if (t->right) q.push(t->right);
    }
    a.push_back(row);
  }
  return a;
}

// 滑动窗口
int lengthOfLongestSubstring(string s) {
  vector<int> m(256, -1);
  int L = 0, ans = 0;
  for (int R = 0; R < (int)s.size(); ++R) {
    if (m[(unsigned char)s[R]] >= L) L = m[(unsigned char)s[R]] + 1;
    m[(unsigned char)s[R]] = R;
    ans = max(ans, R - L + 1);
  }
  return ans;
}

#### 排序 · 快速排序

```cpp
int part(vector<int>& a, int l, int r) {
  int p = a[r], i = l;
  for (int j = l; j < r; ++j)
    if (a[j] < p) swap(a[i++], a[j]);
  swap(a[i], a[r]);
  return i;
}
void qs(vector<int>& a, int l, int r) {
  if (l >= r) return;
  int m = part(a, l, r);
  qs(a, l, m - 1);
  qs(a, m + 1, r);
}
```

#### 排序 · 堆排序

```cpp
void adjust(vector<int>& a, int l, int r) {
  int pi = a[l], k = l;
  for (int i = 2 * l + 1; i <= r; i = 2 * i + 1) {
    if (i < r && a[i] < a[i + 1]) i++;
    if (pi >= a[i]) break;
    a[k] = a[i];
    k = i;
  }
  a[k] = pi;
}
void heapSort(vector<int>& a) {
  int n = (int)a.size();
  for (int i = n / 2; i >= 0; --i) adjust(a, i, n - 1);
  for (int i = n - 1; i > 0; --i) {
    swap(a[0], a[i]);
    adjust(a, 0, i - 1);
  }
}
```

int knap01(const vector<int>& w, const vector<int>& v, int C) {
  vector<int> dp(C + 1, 0);
  for (size_t i = 0; i < w.size(); ++i)
    for (int j = C; j >= w[i]; --j)
      dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
  return dp[C];
}

#### 回溯 · 全排列

```cpp
void permute(vector<int>& a, int k, vector<vector<int>>& o) {
  if (k == (int)a.size()) { o.push_back(a); return; }
  for (int i = k; i < (int)a.size(); ++i) {
    swap(a[k], a[i]);
    permute(a, k + 1, o);
    swap(a[k], a[i]);
  }
}
```

#### 回溯 · 全排列（含重复元素）

```cpp
void permU(const vector<int>& a, vector<int>& p, vector<bool>& u, vector<vector<int>>& o) {
  if (p.size() == a.size()) { o.push_back(p); return; }
  for (int i = 0; i < (int)a.size(); ++i) {
    if (u[i]) continue;
    if (i > 0 && a[i] == a[i - 1] && !u[i - 1]) continue;
    u[i] = true;
    p.push_back(a[i]);
    permU(a, p, u, o);
    p.pop_back();
    u[i] = false;
  }
}
```

void comb(vector<int>& a, int t, int s, int k, vector<int>& p, vector<vector<int>>& o) {
  if (s == t) { o.push_back(p); return; }
  if (s > t) return;
  for (int i = k; i < (int)a.size(); ++i) {
    p.push_back(a[i]);
    comb(a, t, s + a[i], i, p, o);
    p.pop_back();
  }
}

int minRemoval(vector<vector<int>>& a) {
  if (a.empty()) return 0;
  sort(a.begin(), a.end(), [](const auto& x, const auto& y) {
    return x[1] < y[1];
  });
  int keep = 1, e = a[0][1];
  for (size_t i = 1; i < a.size(); ++i) {
    if (a[i][0] >= e) { ++keep; e = a[i][1]; }
  }
  return (int)a.size() - keep;
}

#### 二分查找 · 搜索旋转排序数组

```cpp
int search(vector<int>& a, int t) {
  int l = 0, r = a.size() - 1;
  while (l <= r) {
    int m = l + (r - l) / 2;
    if (a[m] == t) return m;
    if (a[l] <= a[m]) {
      if (a[l] <= t && t < a[m]) r = m - 1; else l = m + 1;
    } else {
      if (a[m] < t && t <= a[r]) l = m + 1; else r = m - 1;
    }
  }
  return -1;
}
```

int findPeakElement(vector<int>& a) {
  int l = 0, r = a.size() - 1;
  while (l < r) {
    int m = l + (r - l) / 2;
    if (a[m] < a[m + 1]) l = m + 1; else r = m;

}
