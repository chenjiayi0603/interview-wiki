# 4. 解决方案：使用 weak_ptr

## 4.1 正确代码实现

```cpp
// 正确代码：使用 weak_ptr 避免循环引用 ✅
class PhoneDisplay : public Observer {
private:
    std::string name_;
    std::weak_ptr<Subject> subject_;  // ← 使用 weak_ptr 而不是 shared_ptr

public:
    PhoneDisplay(const std::string& name, std::shared_ptr<Subject> subject) 
        : name_(name), subject_(subject) {}
    
    void update(const std::string& message) override {
        std::cout << "PhoneDisplay [" << name_ << "] received: " 
                  << message << std::endl;
    }
    
    void unsubscribe() {
        if (auto subject = subject_.lock()) {  // ← 使用 lock() 获取 shared_ptr
            subject->detach(shared_from_this());
        }
    }
};

// Subject 也需要使用 weak_ptr 存储 Observer
class WeatherStation : public Subject {
private:
    std::vector<std::weak_ptr<Observer>> observers_;  // ← 使用 weak_ptr
    float temperature_;

public:
    // 添加观察者
    void attach(std::shared_ptr<Observer> observer) override {
        observers_.push_back(observer);  // weak_ptr 可以从 shared_ptr 构造
    }

    // 移除观察者
    void detach(std::shared_ptr<Observer> observer) override {
        observers_.erase(
            std::remove_if(observers_.begin(), observers_.end(),
                [&observer](const std::weak_ptr<Observer>& wp) {
                    auto locked = wp.lock();
                    // 如果对象已销毁，也移除（清理空的 weak_ptr）
                    if (!locked) {
                        return true;  // 移除空的 weak_ptr
                    }
                    // 如果对象还存在，检查是否匹配
                    return locked == observer;
                }),
            observers_.end()
        );
    }
    
    // 通知所有观察者
    void notify(const std::string& message) override {
        for (auto it = observers_.begin(); it != observers_.end();) {
            if (auto observer = it->lock()) {  // ← 使用 lock() 获取 shared_ptr
                // 对象还存在，通知它
                observer->update(message);
                ++it;
            } else {
                // Observer 已被销毁，移除无效的 weak_ptr
                it = observers_.erase(it);
            }
        }
    }

    // 设置温度并通知观察者
    void setTemperature(float temp) {
        temperature_ = temp;
        notify("Temperature updated to: " + std::to_string(temp));
    }
};
```
