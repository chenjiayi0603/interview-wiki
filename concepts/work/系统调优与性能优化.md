```cpp
void optimizeSocket(int sockfd) {
    // 1. 禁用Nagle算法
    int flag = 1;
    setsockopt(sockfd, IPPROTO_TCP, TCP_NODELAY, &flag, sizeof(flag));
    
    // 2. 禁用延迟确认
    setsockopt(sockfd, IPPROTO_TCP, TCP_QUICKACK, &flag, sizeof(flag));
    
    // 3. 设置发送/接收缓冲区
    int bufsize = 1024 * 1024;  // 1MB
    setsockopt(sockfd, SOL_SOCKET, SO_RCVBUF, &bufsize, sizeof(bufsize));
    setsockopt(sockfd, SOL_SOCKET, SO_SNDBUF, &bufsize, sizeof(bufsize));
    
    // 4. 设置SO_REUSEADDR
    setsockopt(sockfd, SOL_SOCKET, SO_REUSEADDR, &flag, sizeof(flag));
    
    // 5. 设置SO_REUSEPORT（Linux 3.9+）
    setsockopt(sockfd, SOL_SOCKET, SO_REUSEPORT, &flag, sizeof(flag));
}
```

### CPU绑核与NUMA优化

**CPU绑核：**
```cpp
#include <sched.h>
#include <pthread.h>

// 绑定线程到指定CPU核心
void bindThreadToCPU(int cpu_id) {
    cpu_set_t cpuset;
    CPU_ZERO(&cpuset);
    CPU_SET(cpu_id, &cpuset);
    
    pthread_t thread = pthread_self();
    pthread_setaffinity_np(thread, sizeof(cpuset), &cpuset);
}

// 绑定进程到指定CPU核心
void bindProcessToCPU(int cpu_id) {
    cpu_set_t cpuset;
    CPU_ZERO(&cpuset);
    CPU_SET(cpu_id, &cpuset);
    
    pid_t pid = getpid();
    sched_setaffinity(pid, sizeof(cpuset), &cpuset);
}

// 高频交易中的应用：关键线程绑定到独立CPU
void setupTradingThreads() {
    // 订单处理线程绑定到CPU 0
    std::thread order_thread([] {
        bindThreadToCPU(0);
        processOrders();
    });
    
    // 行情处理线程绑定到CPU 1
    std::thread market_thread([] {
        bindThreadToCPU(1);
        processMarketData();
    });
}
```

```cpp
#include <numa.h>

// 获取NUMA节点
int getNUMANode(int cpu_id) {
    return numa_node_of_cpu(cpu_id);
}

// 在指定NUMA节点分配内存
void* allocateOnNUMA(size_t size, int node) {
    return numa_alloc_onnode(size, node);
}

// 绑定线程到NUMA节点
void bindToNUMANode(int node) {
    struct bitmask* nodemask = numa_allocate_nodemask();
    numa_bitmask_setbit(nodemask, node);
    numa_bind(nodemask);
    numa_free_nodemask(nodemask);
}

// 高频交易中的应用：数据本地性优化
class NUMAOptimizedOrderBook {
    int numa_node;
    void* order_buffer;
    
public:
    NUMAOptimizedOrderBook(int cpu_id) {
        numa_node = getNUMANode(cpu_id);
        bindToNUMANode(numa_node);
        order_buffer = allocateOnNUMA(1024 * 1024, numa_node);
    }
};
```

```bash
# 为每个接收队列设置RPS CPU掩码
echo f > /sys/class/net/eth0/queues/rx-0/rps_cpus  # CPU 0-3
echo f0 > /sys/class/net/eth0/queues/rx-1/rps_cpus  # CPU 4-7
```

**RFS (Receive Flow Steering)：**
```bash
# 设置RFS表大小
```

### SIMD指令加速计算

```cpp
#include <immintrin.h>

// AVX并行计算：每次处理8个double
void calculateOrderValue_avx(const double* prices, const double* quantities, 
                             double* values, size_t n) {
    size_t i = 0;
    for (; i + 4 <= n; i += 4) {  // AVX处理4个double
        __m256d p = _mm256_loadu_pd(&prices[i]);
        __m256d q = _mm256_loadu_pd(&quantities[i]);
        __m256d v = _mm256_mul_pd(p, q);
        _mm256_storeu_pd(&values[i], v);
    }
    // 处理剩余元素
    for (; i < n; ++i) {
        values[i] = prices[i] * quantities[i];
    }
}

// 高频交易中的应用：批量计算订单价值
void batchCalculateOrderValues(const std::vector<Order>& orders) {
    const size_t n = orders.size();
    std::vector<double> prices(n), quantities(n), values(n);
    
    // 提取数据到SoA格式
    for (size_t i = 0; i < n; ++i) {
        prices[i] = orders[i].price;
        quantities[i] = orders[i].quantity;
    }
    
    // SIMD加速计算
    calculateOrderValue_avx(prices.data(), quantities.data(), 
                           values.data(), n);
}
```
