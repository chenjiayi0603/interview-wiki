# 事件驱动架构

> 事件驱动架构是一种基于事件的生产、检测、消费和响应的软件架构范式，在低延迟系统中通过状态机实现高效的事件处理。

See also: [[设计模式]], [[C++多线程与并发]]

## 基于状态机的事件处理

```cpp
// 基于状态机的事件处理
class OrderStateMachine {
public:
    enum class State {
        IDLE,
        VALIDATING,
        RISK_CHECKING,
        SENDING,
        PENDING,
        FILLED,
        REJECTED
    };
    
    enum class Event {
        NEW_ORDER,
        VALIDATION_PASS,
        VALIDATION_FAIL,
        RISK_PASS,
        RISK_FAIL,
        ACK_RECEIVED,
        FILL_RECEIVED
    };
    
    using Handler = void(*)(OrderContext&);
    
    // 状态转换表（编译期确定，缓存友好）
    static constexpr State transition_table[7][7] = {
        // NEW_ORDER, VAL_PASS, VAL_FAIL, RISK_PASS, RISK_FAIL, ACK, FILL
        {VALIDATING,  IDLE,     IDLE,     IDLE,      IDLE,      IDLE,   IDLE},    // IDLE
        {IDLE,        RISK_CHECKING, REJECTED, IDLE, IDLE,    IDLE,   IDLE},    // VALIDATING
        {IDLE,        IDLE,     IDLE,     SENDING,   REJECTED,  IDLE,   IDLE},    // RISK_CHECKING
        {IDLE,        IDLE,     IDLE,     IDLE,      IDLE,      PENDING, IDLE},   // SENDING
        {IDLE,        IDLE,     IDLE,     IDLE,      IDLE,      IDLE,   FILLED},  // PENDING
        {IDLE,        IDLE,     IDLE,     IDLE,      IDLE,      IDLE,   IDLE},    // FILLED
        {IDLE,        IDLE,     IDLE,     IDLE,      IDLE,      IDLE,   IDLE},    // REJECTED
    };
    
    void process_event(OrderContext& ctx, Event event) {
        State current = ctx.state;
        State next = transition_table[static_cast<int>(current)]
                                      [static_cast<int>(event)];
        if (next != current) {
            ctx.state = next;
            execute_action(ctx, next);
        }
    }
};
```
