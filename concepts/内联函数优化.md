# 内联函数优化

## 强制内联

```cpp
// 强制内联关键函数
inline __attribute__((always_inline)) 
int fast_add(int a, int b) {
    return a + b;
}

// 或者使用 C++ 方式
inline int fast_add(int a, int b) __attribute__((always_inline));
```

## 分支预测提示

```cpp
// 编译器提示：likely/unlikely
if (__builtin_expect(condition, 1)) {
    // 大多数情况下为真
}
```

## 完整示例

```cpp
// inline_optimize.cpp - 演示内联和分支预测优化
#include <iostream>
#include <chrono>
#include <random>
#include <vector>

// ==================== 内联优化示例 ====================

// 1. 普通函数（可能不内联）
int normal_add(int a, int b) {
    return a + b;
}

// 2. inline 提示（编译器可能忽略）
inline int inline_add(int a, int b) {
    return a + b;
}

// 3. 强制内联（编译器必须内联，除非不可能）
__attribute__((always_inline)) 
inline int force_inline_add(int a, int b) {
    return a + b;
}

// 4. 禁止内联（用于调试或特殊场景）
__attribute__((noinline))
int noinline_add(int a, int b) {
    return a + b;
}

// ==================== 分支预测示例 ====================

// likely: 告诉编译器条件很可能为真
#define likely(x)   __builtin_expect(!!(x), 1)
// unlikely: 告诉编译器条件很可能为假
#define unlikely(x) __builtin_expect(!!(x), 0)

// C++20 标准属性（如果编译器支持）
// [[likely]] 和 [[unlikely]]

// 无分支预测提示的处理函数
int process_without_hint(int value, bool is_valid) {
    if (is_valid) {
        return value * 2 + 1;
    } else {
        return -1;  // 错误处理路径
    }
}

// 带分支预测提示的处理函数（假设大部分数据有效）
int process_with_likely(int value, bool is_valid) {
    if (likely(is_valid)) {
        return value * 2 + 1;  // 热路径
    } else {
        return -1;  // 冷路径
    }
}

// 带分支预测提示的错误检查（假设很少出错）
int check_and_process(int value) {
    if (unlikely(value < 0)) {
        // 异常处理，很少执行
        std::cerr << "Error: negative value" << std::endl;
        return -1;
    }
    // 正常处理，经常执行
    return value * value;
}

// ==================== 性能测试 ====================

template<typename Func>
long long benchmark_inline(Func func, int iterations) {
    auto start = std::chrono::high_resolution_clock::now();
    long long sum = 0;
    for (int i = 0; i < iterations; ++i) {
        sum += func(i, i + 1);
    }
    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::nanoseconds>(end - start);
    return duration.count();
}

void test_branch_prediction(const std::vector<bool>& conditions, int iterations) {
    auto start = std::chrono::high_resolution_clock::now();
    long long sum = 0;
    for (int iter = 0; iter < iterations; ++iter) {
        for (size_t i = 0; i < conditions.size(); ++i) {
            sum += process_with_likely(static_cast<int>(i), conditions[i]);
        }
    }
    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::microseconds>(end - start);
    std::cout << "带likely提示: " << duration.count() << " 微秒, sum=" << sum << std::endl;
    
    start = std::chrono::high_resolution_clock::now();
    sum = 0;
    for (int iter = 0; iter < iterations; ++iter) {
        for (size_t i = 0; i < conditions.size(); ++i) {
            sum += process_without_hint(static_cast<int>(i), conditions[i]);
        }
    }
    end = std::chrono::high_resolution_clock::now();
    duration = std::chrono::duration_cast<std::chrono::microseconds>(end - start);
    std::cout << "无提示:      " << duration.count() << " 微秒, sum=" << sum << std::endl;
}

int main() {
    const int ITERATIONS = 100000000;
    
    std::cout << "=== 内联优化测试 ===" << std::endl;
    
    // 测试不同内联方式的性能
    std::cout << "普通函数:   " << benchmark_inline(normal_add, ITERATIONS) << " 纳秒" << std::endl;
    std::cout << "inline:     " << benchmark_inline(inline_add, ITERATIONS) << " 纳秒" << std::endl;
    std::cout << "强制内联:   " << benchmark_inline(force_inline_add, ITERATIONS) << " 纳秒" << std::endl;
    std::cout << "禁止内联:   " << benchmark_inline(noinline_add, ITERATIONS) << " 纳秒" << std::endl;
    
    std::cout << "\n=== 分支预测测试 ===" << std::endl;
    
    // 准备测试数据：95%为true，5%为false（模拟大部分数据有效的场景）
    std::vector<bool> mostly_true(10000);
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> dis(1, 100);
    for (auto& b : mostly_true) {
        b = (dis(gen) <= 95);  // 95%概率为true
    }
    
    std::cout << "95%为true的分支测试:" << std::endl;
    test_branch_prediction(mostly_true, 1000);
    
    // 随机分布测试
    std::vector<bool> random_dist(10000);
    for (auto& b : random_dist) {
        b = (dis(gen) <= 50);  // 50%概率为true
    }
    
    std::cout << "\n50%随机分布的分支测试:" << std::endl;
    test_branch_prediction(random_dist, 1000);
    
    return 0;
}
```

## C++20 标准分支提示

```cpp
// C++20 标准方式（更推荐）
int modern_process(int value, bool is_valid) {
    if (is_valid) [[likely]] {
        return value * 2 + 1;
    } else [[unlikely]] {
        return -1;
    }
}

// switch 语句中的使用
int handle_type(int type) {
    switch (type) {
        [[likely]] case 0:      // 最常见情况
            return process_type_0();
        case 1:
            return process_type_1();
        [[unlikely]] case -1:   // 错误情况
            return handle_error();
        default:
            return default_handler();
    }
}
```
