-- 秒杀 Lua 脚本（KEYS[1]=商品ID, ARGV[1]=用户ID, ARGV[2]=数量）
local stock_key = "stock:" .. KEYS[1]
local user_key = "bought:" .. KEYS[1]

-- 防止重复购买
if redis.call("SISMEMBER", user_key, ARGV[1]) == 1 then
    return {err = "duplicate"}
end

local stock = redis.call("GET", stock_key)
if not stock or tonumber(stock) < tonumber(ARGV[2]) then
    return {err = "stock_not_enough"}
end

redis.call("DECRBY", stock_key, ARGV[2])
redis.call("SADD", user_key, ARGV[1])
redis.call("EXPIRE", user_key, 86400)  -- 记录保留 1 天

return {ok = "success"}
```

```go
// Go 调用 Lua 脚本
func TrySeckill(rdb *redis.Client, ctx context.Context, productID, userID string, num int64) error {
    script := redis.NewScript(`
        local stock_key = KEYS[1]
        local user_key  = KEYS[2]
        if redis.call("SISMEMBER", user_key, ARGV[1]) == 1 then
            return redis.error_reply("DUPLICATE")
        end
        local stock = redis.call("GET", stock_key)
        if not stock or tonumber(stock) < tonumber(ARGV[2]) then
            return redis.error_reply("STOCK_NOT_ENOUGH")
        end
        redis.call("DECRBY", stock_key, ARGV[2])
        redis.call("SADD", user_key, ARGV[1])
        redis.call("EXPIRE", user_key, 86400)
        return redis.status_reply("OK")
    `)
    keys := []string{"stock:" + productID, "bought:" + productID}
    err := script.Run(ctx, rdb, keys, userID, num).Err()
    if err != nil {
        if strings.Contains(err.Error(), "DUPLICATE") {
            return ErrDuplicatePurchase
        }
        if strings.Contains(err.Error(), "STOCK_NOT_ENOUGH") {
            return ErrStockNotEnough
        }
        return err
    }
    return nil
}

#### 追问热点库存 → 分片存储

```
热点 key 问题: 单个商品秒杀，所有请求打在同一 Redis 节点

分片方案:
  商品库存 key = "stock:{productID}:{shardID}"
  共 N 个分片 (N=16/32/64)
  扣减时: hash(userID) % N 选择分片
  查询时: 汇总所有分片

优点: 写压力分散到多个 Redis 分片节点
缺点: 汇总读慢 + 可能某个分片先空

优化: 热门商品分片数加大(128/256)，冷门商品 1 个分片
```

#### 完整实现: Redis 分布式锁

```go
// Redis 分布式锁
type RedisLock struct {
    rdb    *redis.Client
    key    string
    value  string // 唯一请求 ID（UUID），防误删
    ttl    time.Duration
    done   chan struct{}
}

func NewRedisLock(rdb *redis.Client, key string) *RedisLock {
    return &RedisLock{
        rdb:   rdb,
        key:   "lock:" + key,
        value: uuid.New().String(),
        ttl:   30 * time.Second,
        done:  make(chan struct{}),
    }
}

// TryLock 加锁（SET NX + EXPIRE 原子操作）
func (l *RedisLock) TryLock(ctx context.Context) (bool, error) {
    ok, err := l.rdb.SetNX(ctx, l.key, l.value, l.ttl).Result()
    if err != nil {
        return false, err
    }
    if ok {
        go l.watchDog() // 启动续期 WatchDog
    }
    return ok, nil
}

// WatchDog 自动续期（每 1/3 TTL 续期一次）
func (l *RedisLock) watchDog() {
    ticker := time.NewTicker(l.ttl / 3)
    defer ticker.Stop()
    for {
        select {
        case <-ticker.C:
            // 使用 Lua 检查 value 是否一致再续期
            script := `
                if redis.call("GET", KEYS[1]) == ARGV[1] then
                    return redis.call("EXPIRE", KEYS[1], ARGV[2])
                else
                    return 0
                end
            `
            l.rdb.Eval(context.Background(), script, []string{l.key}, l.value, int(l.ttl.Seconds()))
        case <-l.done:
            return
        }
    }
}

// Unlock 安全解锁（Lua 保证原子性）
func (l *RedisLock) Unlock(ctx context.Context) error {
    close(l.done)
    script := `
        if redis.call("GET", KEYS[1]) == ARGV[1] then
            return redis.call("DEL", KEYS[1])
        else
            return 0
        end
    `
    return l.rdb.Eval(ctx, script, []string{l.key}, l.value).Err()
}
```

#### 追问: 主从切换锁丢失 → 怎么解决？

```
问题: Redis 主节点宕机 → 从节点未同步锁 → 新主节点无锁 → 两个请求同时获得锁

解决思路（非 Redlock）:
  方案 A: 使用 Redis 高可用 + 锁降级
    ├── Redis Cluster / 哨兵保证主从切换秒级完成
    ├── 锁超时时间设短（如 500ms），主从切换期间锁自动过期
    └── 业务层做幂等（锁丢了最多并发执行两次，不会数据不一致）

  方案 B: 业务锁 + 数据库兜底
    ├── Redis 锁抢到 → 执行业务 → 数据库版本号校验
    ├── 若 Redis 锁丢了（并发写入），数据库乐观锁拦截
    └── 双保险: Redis 保证大部分情况不冲突，DB 保证极端情况不数据错乱

  方案 C（不推荐 Redlock）: ...
```

#### Redlock 争议详解（面试加分项）

```
背景: Redis 作者 antirez 提出 Redlock（5 节点多数派加锁），
      Martin Kleppmann（《DDIA》作者）发文批判。

Martin 的批判:
  - 依赖系统时钟: 节点时钟跳跃 → 锁提前过期 → 两个客户端同时持有锁
  - 不是真正的分布式锁: 
    严格定义下，分布式锁需要 fencing token（单调递增的 token 做防重放），
    Redis 无法提供 fencing token 语义。
  - 结论: 对"锁"的要求极为严苛的场景（如 Google Chubby 保护 GFS 元数据），
    Redlock 不安全。

antirez 的反驳:
  - 时钟跳跃是小概率事件，生产环境有 NTP 不会频繁发生
  - 大多数业务场景不需要 fencing token
  - 没有银弹，Redlock 在"足够好"的范围内可用

实际生产选型:
  ├── 业务锁（秒杀/优惠券/库存扣减）: Redis SETNX 就够了，不需要 Redlock
  │    原因: 锁失败 = 多扣一个库存 / 多发一张券，有幂等和补偿兜底
  ├── 配置锁（leader 选举/配置协调）: 用 Etcd/ZK（强一致 ZAB/Raft）
  │    原因: 锁失败 = 脑裂 / 配置错乱，后果严重
  └── Redlock 在业界用得很少: 复杂度高，收益有限，不如直接用 Etcd

总结:
  Redis 锁不是"不安全"，而是"在合适的场景用合适的锁":
  - **业务锁 → Redis**（简单、快、有补偿）
  - **配置锁 → Etcd/ZK**（强一致、可靠、慢一点可接受）

##### 第一层: 业务幂等（防重复执行）

```
❓ 常见疑问: "缓存 + 数据库唯一约束双检查"怎么检查？

答: 分两步，缓存是"快路径"，数据库是"权威依据"。

  ① 缓存检查（快路径）:
     SETNX idempotent:{key} 1 EX 86400
     如果返回 0 → key 已存在 → 说明已处理过 → 直接返回"重复请求"
     如果返回 1 → 第一次见 → 放行到下一步

  ② 数据库唯一约束（权威依据）:
     INSERT INTO idempotent_record (idempotent_key, ...) VALUES (?)
     idempotent_key 列有 UNIQUE 索引
     如果插入成功 → 确认是第一次处理
     如果 Duplicate entry 异常 → 说明缓存不可靠（如 Redis 宕机导致 key 丢了）
       → 同样返回"重复请求"

  ③ 缓存与数据库的关系:
     缓存不是"先检查再执行"，而是"先标记 + 数据库兜底"。
     即使 Redis 挂了，数据库的 UNIQUE 索引仍然能拦截重复请求。
     先写缓存是为了让 99% 的重复请求在缓存层就被拦下，不落 DB。
```

```sql
-- 幂等表: idempotent_key 是唯一约束
CREATE TABLE idempotent_record (
    idempotent_key VARCHAR(128) PRIMARY KEY,  -- 幂等 key 唯一约束
    biz_type       VARCHAR(32),               -- 业务类型（order/payment/coupon）
    created_at     DATETIME,
    INDEX idx_created_at (created_at)
);

-- 插入时如果 key 重复 → Duplicate entry 异常 → 已处理过
INSERT INTO idempotent_record (idempotent_key, biz_type, created_at)
VALUES (?, 'order', NOW());

-- 订单表: 同样可以用 order_id 做唯一约束
CREATE TABLE `order` (
    order_id    VARCHAR(64) PRIMARY KEY,  -- 业务层生成的唯一 ID = 幂等 key
    user_id     BIGINT,
    amount      DECIMAL(10,2),
    status      TINYINT,
    created_at  DATETIME

```

```go
// 双检查的完整流程
func ProcessOrder(ctx context.Context, req *CreateOrderReq) error {
    idempotentKey := req.IdempotentKey // 例如: "order:user_1001:sku_2001:12345678"

    // ① 缓存检查（快路径）
    ok, err := cache.SetNX(ctx, "idempotent:"+idempotentKey, "1", 24*time.Hour)
    if err != nil {
        // Redis 挂了？没关系，继续走数据库检查
        log.Warn("Redis 不可用，降级到数据库幂等检查")
    } else if !ok {
        return ErrDuplicateRequest // 缓存命中了重复请求
    }

    // ② 数据库检查（权威依据）
    err = db.Exec("INSERT INTO idempotent_record (idempotent_key, biz_type) VALUES (?, 'order')", idempotentKey)
    if err != nil {
        if isDuplicateEntry(err) {
            // 数据库说已处理过 → 回滚缓存标记（防止缓存与 DB 不一致）
            cache.Del(ctx, "idempotent:"+idempotentKey)
            return ErrDuplicateRequest
        }
        return err
    }

    // ③ 真正执行业务（此时 100% 确认是第一次处理）
    return orderService.Create(ctx, req)
}
```

```go
// 业务层: 每个请求携带幂等 token
func CreateOrder(ctx context.Context, req *CreateOrderReq) error {
    // 1. 先检查幂等 token 是否已处理
    existed, _ := cache.Get(ctx, "idempotent:"+req.IdempotentKey)
    if existed != nil {
        return ErrDuplicateRequest
    }

    // 2. Redis 分布式锁（SETNX）
    lock := NewRedisLock(rdb, "order:"+req.UserID)
    ok, err := lock.TryLock(ctx)
    if err != nil || !ok {
        return ErrLockFailed
    }
    defer lock.Unlock(ctx)

    // 3. 执行业务（重复执行也安全）
    err = orderRepo.Create(ctx, req.OrderID, req.UserID, req.Amount)
    if err != nil {
        if isDuplicateEntry(err) {
            // 幂等 token 标记已处理
            cache.Set(ctx, "idempotent:"+req.IdempotentKey, "1", 24*time.Hour)
            return ErrDuplicateRequest
        }
        return err
    }

    // 4. 标记幂等 token
    cache.Set(ctx, "idempotent:"+req.IdempotentKey, "1", 24*time.Hour)
    return nil
}

#### 追问: IdempotentKey 怎么来的？由谁生成？

```
IdempotentKey 是"业务语义上的唯一标识"，不是随便一个 UUID。

常见的生成方式:

  1. 客户端生成（推荐）:
     前端在发起请求时生成，基于业务数据 + 时间戳 + 随机数的哈希。
     优点: 重试时客户端可以带同一个 key，幂等拦截在入口处生效。

  2. 服务端生成:
     基于请求参数的组合做哈希，如 MD5(user_id + action + timestamp)。
     缺点: 参数相同不代表请求相同，时间戳精度不够会误判。

  3. 具体场景举例:

     场景          | IdempotentKey 的构成
     --------------|--------------------
     创建订单      | order_id（订单号本身唯一，下单前先生成）
     支付回调      | payment_id + out_order_id（支付流水号）
     优惠券发放    | user_id + campaign_id（同一活动同一用户只发一次）
     库存扣减      | sku_id + request_id（每次操作带 request_id）

  关键原则: 
    幂等 key = 业务上"相同操作"的唯一定义。
    比如"支付回调"的幂等 key 是 payment_id，因为同一笔支付重复回调
    应该只处理一次。如果用 UUID，每次回调 UUID 不同，幂等就失效了。

  实战代码:
    // 客户端生成 IdempotentKey
    function generateIdempotentKey(orderReq) {
        const raw = `${orderReq.userId}:${orderReq.skuId}:${Date.now()}`;
        return CryptoJS.MD5(raw).toString();
    }

    // 支付回调场景：用支付流水号做幂等 key
    // 支付平台会重试回调，但携带同一个 out_trade_no
    func PaymentCallback(outTradeNo, orderId string) error {
        key := "idempotent:payment:" + outTradeNo
        // 用 Redis SETNX 检查是否已处理
        ok, _ := cache.SetNX(ctx, key, "1", 24*time.Hour)
        if !ok {
            return nil // 已处理过，幂等返回成功
        }
        // 执行业务：更新订单状态
        return orderService.PaySuccess(ctx, orderId)
    }
```

##### 第二层: 数据库乐观锁（兜住并发写入）

```
❓ 常见疑问: WHERE version = ? 是一条 SQL 还是两条？

答: 是 2 步（SELECT + UPDATE），但语义上是一次 CAS 操作。

  第 1 步: SELECT stock, version FROM sku_stock WHERE sku_id = ?  ← 查当前版本
  第 2 步: UPDATE ... WHERE version = ?                          ← CAS 更新

  关键: 第 2 步的 WHERE version = ? 判断了"在我读取之后有没有人改过"。
        affected_rows = 0 意味着 version 变了 → 别人先改了 → 重试。
  这不叫"两条 SQL 有并发问题"，因为 UPDATE 是原子操作，
  WHERE 条件判断和更新在同一个事务/同一个行锁内完成。
```

```sql
-- 库存表: version 字段做乐观锁
-- 第 1 条 SQL：读取当前版本
SELECT stock, version FROM sku_stock WHERE sku_id = ?;
-- 返回: stock=100, version=3

-- 第 2 条 SQL：CAS 更新（原子操作）
UPDATE sku_stock 
SET stock = stock - ?, version = version + 1
WHERE sku_id = ? AND stock >= ? AND version = ?;
-- 如果 version 还是 3 → 更新成功，affected_rows = 1
-- 如果 version 被改成 4 → 更新失败，affected_rows = 0 → 说明有并发冲突
-- 此时回滚 Redis 预占库存
```

```go
// 乐观锁更新，冲突时自动重试
func DeductStock(db *sql.DB, skuID string, num int, maxRetry int) error {
    for i := 0; i < maxRetry; i++ {
        // 第 1 步: 读当前版本
        var stock int
        var version int
        db.QueryRow("SELECT stock, version FROM sku_stock WHERE sku_id = ?", skuID).Scan(&stock, &version)

        if stock < num {
            return ErrStockNotEnough
        }

        // 第 2 步: CAS 更新（原子操作）
        result, err := db.Exec(
            "UPDATE sku_stock SET stock = stock - ?, version = version + 1 WHERE sku_id = ? AND version = ?",
            num, skuID, version,
        )
        if err != nil {
            return err
        }
        affected, _ := result.RowsAffected()
        if affected > 0 {
            return nil // 更新成功
        }
        // affected == 0 → 版本冲突，重试
        time.Sleep(time.Duration(rand.Intn(10)) * time.Millisecond)
    }
    return ErrConcurrentConflict
}
```

##### 第三层: 最终对账（兜住所有遗漏）

```
对账脚本（每天凌晨跑）:
  1. Redis 库存日志 vs DB 订单汇总
  2. 发现不一致 → 自动补偿回滚
  3. 实在无法自动修复 → 发告警人工介入

对账 SQL:
  SELECT 
      log.sku_id,
      SUM(log.change) AS redis_change,       -- Redis 库存变化
      SUM(order.num)  AS db_order_deduct      -- DB 实际扣减
  FROM stock_log log
  LEFT JOIN `order` order ON order.sku_id = log.sku_id 
      AND DATE(order.created_at) = DATE(log.created_at)
  WHERE DATE(log.created_at) = CURDATE()
  GROUP BY log.sku_id
  HAVING redis_change != db_order_deduct;
  -- 不一致 → 按差值补偿 Redis 库存
```

##### 三层拦截效果

```
  请求进入
    ↓
  ┌─────────────────────────────────────┐
  │ Redis 锁（拦截 99.9% 的并发冲突）   │
  │ 锁失效概率 ~0.1%（主从切换/超时）    │
  └─────────────────────────────────────┘
    ↓ 锁失效时 ↓
  ┌─────────────────────────────────────┐
  │ 业务幂等（拦截 99% 的重复请求）      │
  │ 幂等 key 检查，相同请求只执行一次    │
  └─────────────────────────────────────┘
    ↓ 幂等也没拦住时 ↓
  ┌─────────────────────────────────────┐
  │ DB 乐观锁（拦截 99.99% 并发写入）    │
  │ version 不匹配直接拒绝，重试或报错   │
  └─────────────────────────────────────┘
    ↓ 乐观锁也没拦住时（理论上 ~0.00001%）↓
  ┌─────────────────────────────────────┐
  │ 最终对账（凌晨补偿，修复所有遗漏）   │
  │ 自动回滚 or 人工介入                │
  └─────────────────────────────────────┘

面试话术:
  "Redis 锁不是 100% 可靠，但它不必是。
   三层补偿机制让数据不一致的概率降到极低:
   即使 Redis 锁丢了 → 幂等拦截重复 → 
   乐观锁兜住并发 → 对账修复遗漏。
   对业务来说，这已经是'足够好'的保证了。
   只有那种锁失败=丢钱的核心金融场景，才需要上 Etcd 强一致锁。"
```

---

### 📌 面试话术总结 · 三层补偿机制（原话背诵）

```
三层补偿机制

  Redis 锁失效 → 业务幂等 → DB乐观锁 → 最终对账
  (99.9%)        (99%)      (99.99%)    (兜底)

  1. 第一层 - 业务幂等：
     每个请求带唯一 IdempotentKey，缓存 + 数据库唯一约束双检查，相同请求只执行一次

  2. 第二层 - DB 乐观锁：
     stock = stock - ? WHERE version = ?，CAS 更新防并发写入，冲突自动重试

  3. 第三层 - 最终对账：
     凌晨跑对账脚本，对比 Redis 库存日志与 DB 订单汇总，不一致自动补偿修复

  面试时可以这样讲：

  ▎ "Redis 锁不是 100% 可靠，但它不必是。
  ▎ 三层补偿机制让数据不一致的概率降到极低：
  ▎ 锁丢了 → 幂等拦截 → 乐观锁兜住 → 对账修复。
  ▎ 对业务来说已经足够好了。
  ▎ 只有锁失败=丢钱的核心金融场景，才需要上 Etcd 强一致锁。"
```

### Q3: RocketMQ 事务消息怎么实现最终一致性？

```
场景: 下单 + 扣库存 + 优惠券 三步骤一致性
```

#### RocketMQ 事务消息

```java
// RocketMQ 事务消息生产者
public class OrderTransactionProducer {
    public static void main(String[] args) throws Exception {
        TransactionMQProducer producer = new TransactionMQProducer("order_group");
        producer.setNamesrvAddr("127.0.0.1:9876");

        // 设置事务监听器
        producer.setTransactionListener(new OrderTransactionListener());
        producer.start();

        // 发送半消息（half message）
        Order order = new Order("ORDER001", "SKU-1001", 1, 99.9);
        Message msg = new Message("order_topic", "create", JSON.toJSONBytes(order));
        SendResult result = producer.sendMessageInTransaction(msg, order);
        System.out.println("发送结果: " + result);
    }
}

// 事务监听器
class OrderTransactionListener implements TransactionListener {
    @Override
    public LocalTransactionState executeLocalTransaction(Message msg, Object arg) {
        // 阶段1: 半消息已发送，执行本地事务
        try {
            Order order = (Order) arg;
            orderService.createOrder(order);             // 创建订单
            stockService.deductStock(order.getSkuId(), order.getNum()); // 扣库存
            couponService.useCoupon(order.getUserId());  // 用优惠券
            return LocalTransactionState.COMMIT_MESSAGE;
        } catch (Exception e) {
            log.error("本地事务失败", e);
            return LocalTransactionState.ROLLBACK_MESSAGE;
        }
    }

    @Override
    public LocalTransactionState checkLocalTransaction(MessageExt msg) {
        // 阶段2: RocketMQ 回查未决状态
        String orderId = msg.getKeys();
        Order order = orderService.getOrder(orderId);
        if (order != null && order.getStatus() == OrderStatus.PAID) {
            return LocalTransactionState.COMMIT_MESSAGE;
        }
        if (order == null || order.getStatus() == OrderStatus.FAILED) {
            return LocalTransactionState.ROLLBACK_MESSAGE;
        }
        return LocalTransactionState.UNKNOW; // 继续回查
    }
}
```

#### Kafka 替代方案（自建事务表）

```
Kafka 不支持事务消息，可自建事务表实现最终一致性:

  1. 本地业务表 + 事务消息表（同一数据库事务）
  2. 定时任务扫描"未发送"的消息 → 发送到 Kafka
  3. 消费端做幂等（唯一订单号去重）

核心 SQL:
  BEGIN;
    INSERT INTO orders (order_id, user_id, amount) VALUES (?, ?, ?);
    INSERT INTO tx_message (msg_id, topic, payload, status) 
        VALUES (?, 'order_topic', ?, 'PENDING');
  COMMIT;

  -- 定时任务
  SELECT * FROM tx_message WHERE status = 'PENDING' AND retry_times < 3;
  UPDATE tx_message SET status = 'SENT' WHERE id = ?;
```

```
面试话术:
  Q: "RocketMQ 事务消息和本地消息表有什么区别？"
  A: "RocketMQ 事务消息把 half 消息、本地事务执行、回查机制都封装好了，
      业务只需实现 TransactionListener 接口。
      本地消息表是自建方案，用数据库事务写业务表+消息表，定时任务扫。
      区别: RocketMQ 省去了自己写回查和定时任务的成本，
      但引入 MQ 依赖。小团队用本地消息表，大厂用 RocketMQ。"
```

### Q4: 分布式事务 TCC 怎么实现？

#### TCC 核心流程

```
TCC = Try + Confirm + Cancel

场景: 订单服务(下订单) + 账户服务(扣钱) + 库存服务(扣库存)

Try 阶段（资源预留）:
  订单服务: 创建"预订单"状态=PENDING
  账户服务: 冻结金额（funds_frozen += amount）
  库存服务: 预占库存（stock_reserved += num）

Confirm 阶段（确认执行）:
  订单服务: 订单状态 → PAID
  账户服务: 冻结→实际扣除（funds_frozen -= amount, funds_balance -= amount）
  库存服务: 预占→实际扣除（stock_reserved -= num, stock_actual -= num）

Cancel 阶段（回滚释放）:
  订单服务: 订单状态 → CANCELLED
  账户服务: 解冻金额（funds_frozen -= amount）
  库存服务: 释放预占（stock_reserved -= num）
```

#### TCC 代码骨架（Go）

```go
// TCC 事务管理器
type TCCTransaction struct {
    txID    string
    actions []*TCCAction
}

type TCCAction struct {
    Name    string
    Try     func() error
    Confirm func() error
    Cancel  func() error
}

func (t *TCCTransaction) Execute() error {
    // 1. Try 阶段（所有资源预留）
    completed := make([]int, 0)
    for i, action := range t.actions {
        if err := action.Try(); err != nil {
            // 2. 任一 Try 失败 → 回滚已成功的 Try
            for j := len(completed) - 1; j >= 0; j-- {
                t.actions[completed[j]].Cancel()
            }
            return fmt.Errorf("Try 阶段失败: %w", err)
        }
        completed = append(completed, i)
    }

    // 3. Confirm 阶段（异步执行，幂等）
    for _, action := range t.actions {
        if err := action.Confirm(); err != nil {
            // 4. 记录失败日志 + 定时任务重试
            log.Errorf("Confirm 失败 tx=%s action=%s", t.txID, action.Name)
            // 不阻断，继续执行其余 Confirm
        }
    }
    return nil
}
```

#### 追问: TCC vs 事务消息 怎么选？

```
TCC:
  ├── 优势: 强一致性（两阶段），业务灵活
  ├── 劣势: 开发复杂（需实现 Try/Confirm/Cancel），空回滚问题
  └── 适用: 金额敏感场景（支付/转账）

事务消息（RocketMQ）:
  ├── 优势: 开发简单，RocketMQ 内建支持
  ├── 劣势: 最终一致性，有延迟窗口
  └── 适用: 一般业务场景（下单/库存）

阿里内部: 交易核心用 TCC，非核心用事务消息
```

```
面试话术:
  Q: "TCC 和事务消息怎么选？"
  A: "看你对一致性的要求有多高。
      TCC 是强一致（Try 预留 -> Confirm/Cancel 执行），
      但开发成本高，要写 Try/Confirm/Cancel 三个方法，
      还要处理空回滚和幂等问题。
      事务消息是最终一致，开发简单，但延迟窗口内数据可能不一致。
      阿里内部: 交易核心（涉及资金）用 TCC，
      非核心（积分、通知）用事务消息。"
```

### Q5: 分库分表（ShardingSphere / MyCat）怎么做？

#### 分片策略

```
水平分片（按 user_id % 16）:
  ├── user_0, user_1, ..., user_15（16 个库/表）
  ├── 优点: 数据均匀，扩展方便（翻倍成 32）
  └── 缺点: 跨分片查询复杂，扩容需迁移数据

垂直分片（按业务拆分）:
  ├── order_db（订单）、user_db（用户）、product_db（商品）
  ├── 优点: 业务隔离，单库压力小
  └── 缺点: 跨库 join 困难

分片键选择:
  ├── 选查询最频繁的字段（user_id / order_id）
  ├── 避免跨分片查询（带分片键的 SQL 直接路由）
  └── 不得不跨分片 → 汇总层（中间件聚合）
```

#### ShardingSphere 配置

#### 面试话术

```
阿里场景:
  订单表 10 亿级 → 按 user_id % 64 分 64 张表
  查询带 user_id → 直接路由到对应分片（一次查询）
  后台查询（按时间/状态）→ 扫描所有分片 + 合并（慢，但后台可接受）

问题: 扩容怎么办？
  64 张表 → 128 张表
  方案 1: 停服迁移（半夜维护窗口）
  方案 2: 双写 + 灰度切流（不停服）
  方案 3: 一致性哈希环（预先分配虚拟节点，最少迁移数据）

问题: 分布式 ID 怎么生成？
  ├── Snowflake: 64bit = 1bit符号 + 41bit时间戳 + 10bit机器 + 12bit序列
  ├── Redis INCR: 步长 = 分片数（ID % 分片数 = 分片号）
  └── 美团 Leaf: 号段模式（批量取 ID，性能高）
```

#### 多级库存架构

```
展示库存 → Redis String（允许延迟，10s 更新一次）
实时库存 → Redis Lua 原子操作（秒级精确）
最终库存 → MySQL 乐观锁兜底（最终一致性）
```

#### 库存扣减流程

```go
// 拼团下单库存扣减（Go 实现）
type StockService struct {
    rdb *redis.Client
    db  *sql.DB
    mq  *kafka.Producer
}

// DeductStock 多级库存扣减
func (s *StockService) DeductStock(ctx context.Context, req *DeductReq) error {
    // 1. Redis Lua 实时扣减
    err := s.redisDeduct(ctx, req.SkuID, req.Num)
    if err != nil {
        return fmt.Errorf("库存不足: %w", err)
    }

    // 2. 异步创建订单
    order := &Order{
        OrderID:  GenOrderID(req.UserID),
        SkuID:    req.SkuID,
        Num:      req.Num,
        Status:   OrderStatusPending,
    }

    // 发送 MQ 消息，下游异步写 DB
    msg, _ := json.Marshal(order)
    s.mq.SendMessage(&kafka.Message{
        Topic: "order_create",
        Value: msg,
    })

    // 3. 返回成功（前端展示"已下单"）
    return nil
}

// redisDeduct Lua 原子扣减（防超卖）
func (s *StockService) redisDeduct(ctx context.Context, skuID string, num int) error {
    script := `
        local key = KEYS[1]
        local stock = redis.call("GET", key)
        if not stock or tonumber(stock) < tonumber(ARGV[1]) then
            return 0
        end
        redis.call("DECRBY", key, ARGV[1])
        return 1
    `
    result, err := s.rdb.Eval(ctx, script, []string{"stock:" + skuID}, num).Result()
    if err != nil || result.(int64) == 0 {
        return ErrStockNotEnough
    }
    return nil
}

// DB 异步库存扣减（乐观锁兜底）
func (s *StockService) dbDeduct(skuID string, num int) error {
    // 乐观锁: version 字段防止并发超卖
    result, err := s.db.Exec(`
        UPDATE sku_stock 
        SET stock = stock - ?, version = version + 1 
        WHERE sku_id = ? AND stock >= ? AND version = ?
    `, num, skuID, num, currentVersion)

    affected, _ := result.RowsAffected()
    if affected == 0 {
        // 这里触发补偿: 回滚 Redis 库存
        s.rdb.Incr(context.Background(), "stock:"+skuID)
        log.Warn("库存不一致，已补偿 Redis")
    }
    return nil
}
```

#### 追问: 库存回滚怎么保证一致性？

```
场景: Redis 扣了，但 DB 库存不足 → 需要补偿回滚

方案:
  1. Redis 用 Lua 先扣 "预占库存"，不扣 "实际库存"
  2. DB 落单成功 → 预占库存转为实际扣减
  3. DB 落单失败 → 定时任务释放预占库存
  4. 兜底: 最终对账（每天凌晨库存对账脚本）

对账脚本:
  SELECT sku_id, SUM(IF(type='order', -num, 0)) AS order_deduct,
         SUM(IF(type='compensate', +num, 0)) AS compensate
  FROM stock_log WHERE date = CURDATE()
  GROUP BY sku_id
  HAVING order_deduct != compensate;
```

```
