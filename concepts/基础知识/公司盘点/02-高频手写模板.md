## 一、LRU Cache

```cpp
class LRUCache {
    list<pair<int,int>> li;
    unordered_map<int, list<pair<int,int>>::iterator> mp;
    int cap;
public:
    LRUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        if (!mp.count(key)) return -1;
        li.splice(li.begin(), li, mp[key]);     // 移到头部
        return mp[key]->second;
    }
    void put(int key, int value) {
        if (mp.count(key)) {
            li.splice(li.begin(), li, mp[key]);
            mp[key]->second = value;
            return;
        }
        if (li.size() == cap) {
            mp.erase(li.back().first);
            li.pop_back();
        }
        li.push_front({key, value});
        mp[key] = li.begin();
    }
};
```

## 二、反转链表

```cpp
ListNode* reverseList(ListNode* head) {
    ListNode *pre = nullptr, *cur = head;
    while (cur) {
        ListNode* nxt = cur->next;
        cur->next = pre;
        pre = cur;
        cur = nxt;
    }
    return pre;
}
```

## 三、快排

```cpp
void quickSort(vector<int>& nums, int l, int r) {
    if (l >= r) return;
    int pivot = nums[r];                     // 选最右
    int i = l;
    for (int j = l; j < r; j++)
        if (nums[j] <= pivot)
            swap(nums[i++], nums[j]);
    swap(nums[i], nums[r]);                  // pivot 到位
    quickSort(nums, l, i-1);
    quickSort(nums, i+1, r);
}
```

## 四、第 K 大（最小堆）

```cpp
int findKthLargest(vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int x : nums) {
        pq.push(x);
        if (pq.size() > k) pq.pop();
    }
    return pq.top();
}
```

## 五、链表判环 + 入口

```cpp
ListNode *detectCycle(ListNode *head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {                  // 相遇
            fast = head;                     // 快指针回 head
            while (slow != fast) {
                slow = slow->next;
                fast = fast->next;
            }
            return slow;                     // 入口
        }
    }
    return nullptr;
}
```

## 六、有效括号

```cpp
bool isValid(string s) {
    stack<char> st;
    for (char c : s) {
        if (c == '(') st.push(')');
        else if (c == '[') st.push(']');
        else if (c == '{') st.push('}');
        else if (st.empty() || st.top() != c) return false;
        else st.pop();
    }
    return st.empty();
}
```

## 七、全排列（回溯）

```cpp
vector<vector<int>> permute(vector<int>& nums) {
    vector<vector<int>> res;
    function<void(int)> dfs = [&](int start) {
        if (start == nums.size()) { res.push_back(nums); return; }
        for (int i = start; i < nums.size(); i++) {
            swap(nums[start], nums[i]);
            dfs(start + 1);
            swap(nums[start], nums[i]);
        }
    };
    dfs(0);
    return res;
}
```

## 八、最大子序和（Kadane）

```cpp
int maxSubArray(vector<int>& nums) {
    int cur = 0, best = INT_MIN;
    for (int x : nums) {
        cur = max(x, cur + x);
        best = max(best, cur);
    }
    return best;
}
```

## 九、爬楼梯

```cpp
int climbStairs(int n) {
    int a = 1, b = 1;    // dp[0]=1, dp[1]=1
    for (int i = 2; i <= n; i++) {
        int c = a + b;
        a = b; b = c;
    }
    return b;
}
```

## 十、合并两个有序链表

```cpp
ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
    ListNode dummy(0), *p = &dummy;
    while (l1 && l2) {
        if (l1->val < l2->val) { p->next = l1; l1 = l1->next; }
        else { p->next = l2; l2 = l2->next; }
        p = p->next;
    }
    p->next = l1 ? l1 : l2;
    return dummy.next;
}
```

## 十一、二叉树层序遍历

```cpp
vector<vector<int>> levelOrder(TreeNode* root) {
    if (!root) return {};
    vector<vector<int>> res;
    queue<TreeNode*> q{{root}};
    while (!q.empty()) {
        int sz = q.size();
        vector<int> level;
        for (int i = 0; i < sz; i++) {
            auto* node = q.front(); q.pop();
            level.push_back(node->val);
            if (node->left) q.push(node->left);
            if (node->right) q.push(node->right);
        }
        res.push_back(level);
    }
    return res;
}
```
