## 1. 互斥锁（Mutex）

### 基本概念
互斥锁是最常用的同步原语，保证同一时刻只有一个线程可以访问临界区。

### POSIX pthread_mutex

```cpp
#include <pthread.h>

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

// 或动态初始化
pthread_mutex_t mutex;
pthread_mutex_init(&mutex, NULL);

// 加锁
pthread_mutex_lock(&mutex);

// 尝试加锁（非阻塞）
int ret = pthread_mutex_trylock(&mutex);

// 超时加锁
struct timespec ts;
clock_gettime(CLOCK_REALTIME, &ts);
ts.tv_sec += 5;  // 5秒超时
pthread_mutex_timedlock(&mutex, &ts);

// 解锁
pthread_mutex_unlock(&mutex);

// 销毁
pthread_mutex_destroy(&mutex);
```

### 互斥锁类型

```cpp
pthread_mutexattr_t attr;
pthread_mutexattr_init(&attr);

// PTHREAD_MUTEX_NORMAL: 标准互斥锁，重复加锁会死锁
// PTHREAD_MUTEX_ERRORCHECK: 检错互斥锁，重复加锁返回错误
// PTHREAD_MUTEX_RECURSIVE: 递归互斥锁，同一线程可多次加锁
// PTHREAD_MUTEX_DEFAULT: 默认类型
pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);

pthread_mutex_t mutex;
pthread_mutex_init(&mutex, &attr);
```

### C++11 std::mutex

```cpp
#include <mutex>

std::mutex mtx;

void safe_function() {
    mtx.lock();
    // 临界区
    mtx.unlock();
}

// RAII风格（推荐）
void safe_function_raii() {
    std::lock_guard<std::mutex> lock(mtx);
    // 临界区，离开作用域自动解锁
}

// C++17 带模板参数推导
void safe_function_cpp17() {
    std::lock_guard lock(mtx);  // 自动推导类型
    // 临界区
}
```

## 2. 读写锁（Read-Write Lock）

### 适用场景
- 读多写少的场景
- 多个读者可以同时访问
- 写者独占访问

### POSIX pthread_rwlock

```cpp
#include <pthread.h>

pthread_rwlock_t rwlock = PTHREAD_RWLOCK_INITIALIZER;

// 读锁
pthread_rwlock_rdlock(&rwlock);
// ... 读操作
pthread_rwlock_unlock(&rwlock);

// 写锁
pthread_rwlock_wrlock(&rwlock);
// ... 写操作
pthread_rwlock_unlock(&rwlock);

// 尝试加锁
pthread_rwlock_tryrdlock(&rwlock);
pthread_rwlock_trywrlock(&rwlock);
```

### C++14 std::shared_mutex (C++17正式引入)

```cpp
#include <shared_mutex>

std::shared_mutex rw_mutex;

void reader() {
    std::shared_lock<std::shared_mutex> lock(rw_mutex);
    // 读操作，多个读者可同时持有
}

void writer() {
    std::unique_lock<std::shared_mutex> lock(rw_mutex);
    // 写操作，独占访问
}
```

### 读写锁实现原理

```cpp
class SimpleRWLock {
private:
    std::mutex mtx;
    std::condition_variable cv;
    int readers = 0;
    bool writer = false;
    int waiting_writers = 0;

public:
    void read_lock() {
        std::unique_lock<std::mutex> lock(mtx);
        // 写优先：有等待的写者时，读者等待
        cv.wait(lock, [this]{ return !writer && waiting_writers == 0; });
        ++readers;
    }

    void read_unlock() {
        std::unique_lock<std::mutex> lock(mtx);
        --readers;
        if (readers == 0) cv.notify_all();
    }

    void write_lock() {
        std::unique_lock<std::mutex> lock(mtx);
        ++waiting_writers;
        cv.wait(lock, [this]{ return !writer && readers == 0; });
        --waiting_writers;
        writer = true;
    }

    void write_unlock() {
        std::unique_lock<std::mutex> lock(mtx);
        writer = false;
        cv.notify_all();
    }
};
```

## 3. 自旋锁（Spinlock）

### 特点
- 忙等待，不让出CPU
- 适合临界区极短的场景
- 在单核CPU上可能导致死锁

### POSIX pthread_spinlock

```cpp
#include <pthread.h>

pthread_spinlock_t spinlock;
pthread_spin_init(&spinlock, PTHREAD_PROCESS_PRIVATE);

pthread_spin_lock(&spinlock);
// 临界区
pthread_spin_unlock(&spinlock);

pthread_spin_destroy(&spinlock);
```

### C++原子实现自旋锁

```cpp
#include <atomic>

class SpinLock {
private:
    std::atomic_flag flag = ATOMIC_FLAG_INIT;

public:
    void lock() {
        while (flag.test_and_set(std::memory_order_acquire)) {
            // 自旋等待
            // 可以添加 pause 指令减少CPU功耗
            #if defined(__x86_64__) || defined(_M_X64)
            __builtin_ia32_pause();
            #endif
        }
    }

    void unlock() {
        flag.clear(std::memory_order_release);
    }

    bool try_lock() {
        return !flag.test_and_set(std::memory_order_acquire);
    }
};
```

### 改进版自旋锁（带退避）

```cpp
class SpinLockWithBackoff {
private:
    std::atomic<bool> locked{false};

public:
    void lock() {
        int spin_count = 0;
        while (true) {
            // 先检查再尝试CAS，减少总线流量
            if (!locked.load(std::memory_order_relaxed)) {
                if (!locked.exchange(true, std::memory_order_acquire)) {
                    return;
                }
            }
            
            // 指数退避
            if (++spin_count > 1000) {
                std::this_thread::yield();
                spin_count = 0;
            }
        }
    }

    void unlock() {
        locked.store(false, std::memory_order_release);
    }
};
```

## 4. 条件变量（Condition Variable）

### 用途
- 线程间的等待-通知机制
- 必须配合互斥锁使用

### POSIX pthread_cond

```cpp
#include <pthread.h>

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
bool ready = false;

// 等待线程
void* waiter(void*) {
    pthread_mutex_lock(&mutex);
    while (!ready) {  // 使用while防止虚假唤醒
        pthread_cond_wait(&cond, &mutex);
    }
    // 处理数据
    pthread_mutex_unlock(&mutex);
    return nullptr;
}

// 通知线程
void* notifier(void*) {
    pthread_mutex_lock(&mutex);
    ready = true;
    pthread_cond_signal(&cond);    // 唤醒一个等待线程
    // pthread_cond_broadcast(&cond); // 唤醒所有等待线程
    pthread_mutex_unlock(&mutex);
    return nullptr;
}
```

### C++11 std::condition_variable

```cpp
#include <condition_variable>
#include <mutex>

std::mutex mtx;
std::condition_variable cv;
bool ready = false;

void consumer() {
    std::unique_lock<std::mutex> lock(mtx);
    cv.wait(lock, []{ return ready; });  // 带谓词的wait
    // 处理数据
}

void producer() {
    {
        std::lock_guard<std::mutex> lock(mtx);
        ready = true;
    }
    cv.notify_one();  // 通知在锁外进行效率更高
}
```

### 生产者-消费者模型

```cpp
#include <queue>
#include <mutex>
#include <condition_variable>
#include <thread>

template<typename T>
class ThreadSafeQueue {
private:
    std::queue<T> queue;
    mutable std::mutex mtx;
    std::condition_variable not_empty;
    std::condition_variable not_full;
    size_t max_size;

public:
    explicit ThreadSafeQueue(size_t max = 100) : max_size(max) {}

    void push(T value) {
        std::unique_lock<std::mutex> lock(mtx);
        not_full.wait(lock, [this]{ return queue.size() < max_size; });
        queue.push(std::move(value));
        not_empty.notify_one();
    }

    T pop() {
        std::unique_lock<std::mutex> lock(mtx);
        not_empty.wait(lock, [this]{ return !queue.empty(); });
        T value = std::move(queue.front());
        queue.pop();
        not_full.notify_one();
        return value;
    }

    bool try_pop(T& value, std::chrono::milliseconds timeout) {
        std::unique_lock<std::mutex> lock(mtx);
        if (!not_empty.wait_for(lock, timeout, [this]{ return !queue.empty(); })) {
            return false;
        }
        value = std::move(queue.front());
        queue.pop();
        not_full.notify_one();
        return true;
    }

    bool empty() const {
        std::lock_guard<std::mutex> lock(mtx);
        return queue.empty();
    }

    size_t size() const {
        std::lock_guard<std::mutex> lock(mtx);
        return queue.size();
    }
};

// 使用示例：2 生产者 2 消费者，阻塞式 push/pop（满则等、空则等）
void producer(ThreadSafeQueue<int>& q, int beg, int end) {
    for (int i = beg; i < end; ++i) q.push(i);
}
void consumer(ThreadSafeQueue<int>& q, int n) {
    for (int i = 0; i < n; ++i) q.pop();
}
int main() {
    ThreadSafeQueue<int> q(10);
    std::thread p1(producer, std::ref(q), 0, 10), p2(producer, std::ref(q), 10, 20);
    std::thread c1(consumer, std::ref(q), 10), c2(consumer, std::ref(q), 10);
    p1.join(); p2.join(); c1.join(); c2.join();
    return 0;
}
```

## 5. 信号量（Semaphore）

### POSIX信号量

```cpp
#include <semaphore.h>

sem_t sem;

// 初始化，初始值为1
sem_init(&sem, 0, 1);  // 第二个参数0表示线程间，非0表示进程间

// P操作（等待/减少）
sem_wait(&sem);        // 阻塞等待
sem_trywait(&sem);     // 非阻塞

// 超时等待
struct timespec ts;
clock_gettime(CLOCK_REALTIME, &ts);
ts.tv_sec += 5;
sem_timedwait(&sem, &ts);

// V操作（释放/增加）
sem_post(&sem);

// 获取当前值
int value;
sem_getvalue(&sem, &value);

// 销毁
sem_destroy(&sem);
```

### C++20 std::counting_semaphore

```cpp
#include <semaphore>

// 计数信号量
std::counting_semaphore<10> sem(5);  // 最大10，初始5

sem.acquire();       // P操作
sem.release();       // V操作
sem.try_acquire();   // 非阻塞

// 二元信号量
std::binary_semaphore bsem(1);
bsem.acquire();
bsem.release();
```

### 信号量实现（C++11兼容）

```cpp
#include <mutex>
#include <condition_variable>

class Semaphore {
private:
    std::mutex mtx;
    std::condition_variable cv;
    int count;

public:
    explicit Semaphore(int initial = 0) : count(initial) {}

    void acquire() {
        std::unique_lock<std::mutex> lock(mtx);
        cv.wait(lock, [this]{ return count > 0; });
        --count;
    }

    bool try_acquire() {
        std::lock_guard<std::mutex> lock(mtx);
        if (count > 0) {
            --count;
            return true;
        }
        return false;
    }

    template<class Rep, class Period>
    bool try_acquire_for(const std::chrono::duration<Rep, Period>& timeout) {
        std::unique_lock<std::mutex> lock(mtx);
        if (!cv.wait_for(lock, timeout, [this]{ return count > 0; })) {
            return false;
        }
        --count;
        return true;
    }

    void release(int n = 1) {
        {
            std::lock_guard<std::mutex> lock(mtx);
            count += n;
        }
        if (n == 1) {
            cv.notify_one();
        } else {
            cv.notify_all();
        }
    }
};
```

## 6. 屏障（Barrier）

### 用途
- 让多个线程在某个点等待，直到所有线程都到达

### POSIX pthread_barrier

```cpp
#include <pthread.h>

pthread_barrier_t barrier;
pthread_barrier_init(&barrier, NULL, 3);  // 3个线程

void* thread_func(void* arg) {
    // 第一阶段工作
    
    pthread_barrier_wait(&barrier);  // 等待所有线程到达
    
    // 第二阶段工作
    return nullptr;
}
```

### C++20 std::barrier

```cpp
#include <barrier>
#include <thread>
#include <vector>

void parallel_work() {
    const int num_threads = 4;
    
    // 完成回调函数
    auto on_completion = []() noexcept {
        std::cout << "All threads completed phase\n";
    };
    
    std::barrier sync_point(num_threads, on_completion);
    
    auto worker = [&](int id) {
        // 阶段1
        std::cout << "Thread " << id << " phase 1\n";
        sync_point.arrive_and_wait();
        
        // 阶段2
        std::cout << "Thread " << id << " phase 2\n";
        sync_point.arrive_and_wait();
    };
    
    std::vector<std::thread> threads;
    for (int i = 0; i < num_threads; ++i) {
        threads.emplace_back(worker, i);
    }
    
    for (auto& t : threads) {
        t.join();
    }
}
```

### C++11实现Barrier

```cpp
class Barrier {
private:
    std::mutex mtx;
    std::condition_variable cv;
    const size_t threshold;
    size_t count;
    size_t generation;

public:
    explicit Barrier(size_t count) 
        : threshold(count), count(count), generation(0) {}

    void wait() {
        std::unique_lock<std::mutex> lock(mtx);
        size_t gen = generation;
        
        if (--count == 0) {
            ++generation;
            count = threshold;
            cv.notify_all();
        } else {
            cv.wait(lock, [this, gen]{ return gen != generation; });
        }
    }
};
```

## 7. Latch（门闩）

### C++20 std::latch

```cpp
#include <latch>
#include <thread>

void example_latch() {
    std::latch work_done(3);
    
    auto worker = [&](int id) {
        // 执行工作
        std::cout << "Worker " << id << " done\n";
        work_done.count_down();
    };
    
    std::thread t1(worker, 1);
    std::thread t2(worker, 2);
    std::thread t3(worker, 3);
    
    work_done.wait();  // 等待所有工作完成
    std::cout << "All workers finished\n";
    
    t1.join(); t2.join(); t3.join();
}
```
