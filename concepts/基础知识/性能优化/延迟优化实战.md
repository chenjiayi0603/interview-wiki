## 7.1 p99/p999 延迟分析

### Q25: 如何统计和分析 p99/p999 延迟？

**答案：**

```cpp
#include <vector>
#include <algorithm>
#include <chrono>

class LatencyStatistics {
    std::vector<uint64_t> latencies;  // 微秒
    std::mutex mtx;
    
public:
    void record(uint64_t latency_us) {
        std::lock_guard<std::mutex> lock(mtx);
        latencies.push_back(latency_us);
    }
    
    struct Stats {
        uint64_t p50, p90, p95, p99, p999;
        uint64_t min, max, avg;
    };
    
    Stats calculate() {
        std::lock_guard<std::mutex> lock(mtx);
        if (latencies.empty()) return {};
        
        std::vector<uint64_t> sorted = latencies;
        std::sort(sorted.begin(), sorted.end());
        
        Stats stats;
        size_t n = sorted.size();
        
        stats.min = sorted[0];
        stats.max = sorted[n - 1];
        
        // 计算平均值
        uint64_t sum = 0;
        for (auto l : sorted) sum += l;
        stats.avg = sum / n;
        
        // 计算百分位数
        stats.p50 = sorted[n * 0.50];
        stats.p90 = sorted[n * 0.90];
        stats.p95 = sorted[n * 0.95];
        stats.p99 = sorted[n * 0.99];
        stats.p999 = sorted[n * 0.999];
        
        return stats;
    }
    
    void reset() {
        std::lock_guard<std::mutex> lock(mtx);
        latencies.clear();
    }
};

// 使用示例
class OrderLatencyTracker {
    LatencyStatistics stats;
    
public:
    void processOrder(const Order& order) {
        auto start = std::chrono::high_resolution_clock::now();
        
        // 处理订单
        executeOrder(order);
        
        auto end = std::chrono::high_resolution_clock::now();
        auto latency = std::chrono::duration_cast<std::chrono::microseconds>(
            end - start).count();
        
        stats.record(latency);
    }
    
    void report() {
        auto s = stats.calculate();
        printf("Latency Stats:\n");
        printf("  min: %lu us\n", s.min);
        printf("  p50: %lu us\n", s.p50);
        printf("  p90: %lu us\n", s.p90);
        printf("  p95: %lu us\n", s.p95);
        printf("  p99: %lu us\n", s.p99);
        printf("  p999: %lu us\n", s.p999);
        printf("  max: %lu us\n", s.max);
        printf("  avg: %lu us\n", s.avg);
    }
};
```
