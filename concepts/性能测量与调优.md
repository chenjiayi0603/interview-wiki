# 性能测量与调优

## 7.1 高精度计时

```cpp
#include <chrono>
#include <x86intrin.h>

// 使用 RDTSC 指令（最低开销，~20 cycles）
inline uint64_t rdtsc() {
    return __rdtsc();
}

// 使用 RDTSCP（序列化，更准确）
inline uint64_t rdtscp() {
    unsigned int aux;
    return __rdtscp(&aux);
}

// 内存屏障 + RDTSC
inline uint64_t rdtsc_serialized() {
    _mm_lfence();  // 加载屏障
    return __rdtsc();
}

// 纳秒级计时器类
class NanoTimer {
    uint64_t start_;
    static double cycles_per_ns_;
    
public:
    static void calibrate() {
        auto t1 = std::chrono::high_resolution_clock::now();
        uint64_t c1 = rdtsc();
        
        // 睡眠 100ms 校准
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        
        uint64_t c2 = rdtsc();
        auto t2 = std::chrono::high_resolution_clock::now();
        
        auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(t2 - t1).count();
        cycles_per_ns_ = static_cast<double>(c2 - c1) / ns;
    }
    
    void start() { start_ = rdtsc(); }
    
    uint64_t elapsed_ns() const {
        return static_cast<uint64_t>((rdtsc() - start_) / cycles_per_ns_);
    }
};
```

## 7.2 延迟直方图

```cpp
#include <array>
#include <atomic>

class LatencyHistogram {
    // 对数分桶：1μs, 2μs, 4μs, 8μs, 16μs, ...
    static constexpr int BUCKETS = 32;
    std::array<std::atomic<uint64_t>, BUCKETS> buckets_{};
    
public:
    void record(uint64_t nanoseconds) {
        int bucket = 0;
        uint64_t threshold = 1000;  // 1μs
        
        while (nanoseconds > threshold && bucket < BUCKETS - 1) {
            threshold <<= 1;
            ++bucket;
        }
        
        buckets_[bucket].fetch_add(1, std::memory_order_relaxed);
    }
    
    void print() const {
        uint64_t threshold = 1000;
        for (int i = 0; i < BUCKETS; ++i) {
            uint64_t count = buckets_[i].load(std::memory_order_relaxed);
            if (count > 0) {
                printf("<%luμs: %lu\n", threshold / 1000, count);
            }
            threshold <<= 1;
        }
    }
};
```

## 7.3 Perf 性能分析

```bash
# CPU 性能计数器分析
perf stat -e cycles,instructions,cache-misses,cache-references ./hft_app

# 热点函数分析
perf record -g ./hft_app
perf report

# 缓存未命中分析
perf stat -e L1-dcache-load-misses,L1-dcache-loads ./hft_app

# 分支预测分析
perf stat -e branches,branch-misses ./hft_app

# PMU 事件分析（Intel）
perf stat -e r412e,r4f2e ./hft_app  # L1D 替换, L3 未命中
```
