# 无锁编程

> 无锁编程（Lock-Free Programming）是一种不使用传统互斥锁（mutex）等阻塞同步原语，而通过原子操作（如 CAS）实现线程安全的多线程编程范式。核心目标是避免死锁、优先级反转等问题，并提升高并发场景下的性能。

See also: [[C++多线程与并发]], [[原子操作与内存模型]], [[MPMC环形无锁队列-Vyukov]], [[同步问题与解决方案]]

## 无锁栈

```cpp
#include <atomic>
#include <memory>

template<typename T>
class LockFreeStack {
private:
    struct Node {
        T data;
        Node* next;
        Node(const T& d) : data(d), next(nullptr) {}
    };
    
    std::atomic<Node*> head{nullptr};

public:
    void push(const T& data) {
        Node* new_node = new Node(data);
        new_node->next = head.load(std::memory_order_relaxed);
        
        // CAS循环
        while (!head.compare_exchange_weak(
            new_node->next, new_node,
            std::memory_order_release,
            std::memory_order_relaxed)) {
            // new_node->next被更新为当前head
        }
    }
    
    bool pop(T& result) {
        Node* old_head = head.load(std::memory_order_relaxed);
        
        while (old_head && !head.compare_exchange_weak(
            old_head, old_head->next,
            std::memory_order_acquire,
            std::memory_order_relaxed)) {
            // old_head被更新为当前head
        }
        
        if (!old_head) return false;
        
        result = old_head->data;
        // 注意：这里有内存泄漏风险，实际需要引用计数或风险指针
        delete old_head;
        return true;
    }
};
```

## 无锁队列（单生产者单消费者）

```cpp
#include <atomic>
#include <array>

template<typename T, size_t Size>
class SPSCQueue {
private:
    std::array<T, Size> buffer;
    std::atomic<size_t> head{0};
    std::atomic<size_t> tail{0};

public:
    bool push(const T& item) {
        size_t current_tail = tail.load(std::memory_order_relaxed);
        size_t next_tail = (current_tail + 1) % Size;
        
        if (next_tail == head.load(std::memory_order_acquire)) {
            return false;  // 队列满
        }
        
        buffer[current_tail] = item;
        tail.store(next_tail, std::memory_order_release);
        return true;
    }
    
    bool pop(T& item) {
        size_t current_head = head.load(std::memory_order_relaxed);
        
        if (current_head == tail.load(std::memory_order_acquire)) {
            return false;  // 队列空
        }
        
        item = buffer[current_head];
        head.store((current_head + 1) % Size, std::memory_order_release);
        return true;
    }
};
/*
为啥只能单线程（单生产者单消费者，SPSC，Single Producer Single Consumer）？
原因：
- 这种环形队列实现里，只有一个线程负责push（写入），一个线程负责pop（读取），读写指针各归一方所有，无需多线程同步。
- head（读指针）只会被消费者线程修改，tail（写指针）只会被生产者线程修改，俩线程互不干扰，天然线程安全。
- 不加锁能高效完成数据传递，但如果有多个生产者/消费者同时push/pop就会有数据竞争，需要更复杂的同步方案。

如果多生产者或多消费者：
- 多生产者要安全地修改tail，多消费者要安全地修改head，都需要加锁或CAS等同步机制，否则会破坏队列的一致性。

结论：本实现只保证SPSC，扩展到MPSC、SPMC、MPMC需要更复杂的原子操作或锁。
*/
```

## 多生产多消费（MPMC）环形无锁队列-Vyukov

```cpp
/*
多生产多消费（MPMC）环形无锁队列核心思路：
- 不能再假设只有head/only consumer线程和tail/only producer线程能访问和修改自己的指针。
- 需为每个槽位增加标志（称为sequence/cycle/ticket等）辅助判断该槽当前处于"可写"还是"可读"。
- push/pop都需CAS交换和循环尝试，才能保证多线程下的一致性。

经典的方案是Vyukov的无锁环形队列（bounded MPMC queue），如下：

*/
#include <atomic>
#include <cstddef>
#include <array>

template<typename T, size_t Size>
class MPMCQueue {
    static_assert((Size & (Size - 1)) == 0, "Size must be a power of 2");
    struct Cell {
        std::atomic<size_t> seq;
        T data;
    };
    std::array<Cell, Size> buffer;
    std::atomic<size_t> head{0};  // 64位序号，正常不回绕；回绕时仅容量损失，无脏读写
    std::atomic<size_t> tail{0};
public:
    MPMCQueue() {
        for (size_t i = 0; i < Size; ++i) {
            buffer[i].seq.store(i, std::memory_order_relaxed);
        }
    }

    bool push(const T& data) {
        size_t pos = tail.load(std::memory_order_relaxed);
        for (;;) {
            Cell& cell = buffer[pos & (Size - 1)];
            size_t seq = cell.seq.load(std::memory_order_acquire);
         
            intptr_t dif = (intptr_t)seq - (intptr_t)pos;
            // dif==0：可写；写后 seq=pos+1；消费后 seq=pos+Size，tail 回卷时仍可匹配
            if (dif == 0) {
                if (tail.compare_exchange_weak(pos, pos + 1, std::memory_order_acq_rel, std::memory_order_acquire)) {
                    cell.data = data;
                    cell.seq.store(pos + 1, std::memory_order_release);
                    return true;
                }
            } else if (dif < 0) {  // seq < pos：槽未消费，队列满
                return false;
            } else {  // dif>0：CAS 竞争失败，重试
                pos = tail.load(std::memory_order_relaxed);
            }
        }
    }

    bool pop(T& data) {
        size_t pos = head.load(std::memory_order_relaxed);
        for (;;) {
            Cell& cell = buffer[pos & (Size - 1)];
            size_t seq = cell.seq.load(std::memory_order_acquire);
            intptr_t dif = (intptr_t)seq - (intptr_t)(pos + 1);
            if (dif == 0) {
                if (head.compare_exchange_weak(pos, pos + 1, std::memory_order_acq_rel, std::memory_order_acquire)) {
                    data = cell.data;
                    cell.seq.store(pos + Size, std::memory_order_release);  // +Size 防 ABA，跳到下一轮
                    return true;
                }
            } else if (dif < 0) {  // seq < pos+1：槽未填充，队列空
                return false;
            } else {
                // dif>0：槽已被其他消费者处理，重试
                pos = head.load(std::memory_order_acquire);
            }
        }
    }
};

// Vyukov 无锁 MPMC 队列：seq 表示槽位状态（可写 seq==pos，可读 seq==pos+1，消费后 seq=pos+Size）。
// Size 须为 2 的幂；初始化 cell.seq=i。参考：1024cores.net/articles/queues/bounded-mpmc-queue
```

## 无锁数组队列（多生产者多消费者）-已生产标志

```cpp
// 无锁数组队列（多生产者多消费者），高性能，缓存友好
// 只在缓冲区加入已生产标志，不改变原有其他成员和入队/出队接口逻辑
#include <array>
#include <atomic>
#include <cstddef>

template<typename T, size_t Size>
class LockFreeArrayQueue {
    static_assert((Size & (Size - 1)) == 0, "Size must be a power of 2");
private:
    struct Cell {
        T value;
        std::atomic<bool> produced{false}; // 已生产标志
    };

    std::atomic<size_t> w_{0};   // 写入位置
    std::atomic<size_t> r_{0};   // 读取位置

    std::array<Cell, Size> buf_; // 改为 std::array<Cell, Size>

    size_t mask(size_t pos) const {
        // 取模操作，将pos映射到[0, Size - 1]区间
        return pos & (Size - 1);
    }
public:
    LockFreeArrayQueue() = default;

    // 支持多生产者多消费者的（MPMC）。
    bool enqueue(const T& item) {
        size_t cw, nw;
        do {
            cw = w_.load(std::memory_order_acquire);
            nw = mask(cw + 1); // 使用mask保证下标环绕
            if (nw == r_.load(std::memory_order_acquire))
                return false;  // 队列已满
            // 只有拿到写入权线程继续
        } while (!w_.compare_exchange_weak(cw, nw,
                                           std::memory_order_acq_rel,
                                           std::memory_order_acquire));
        buf_[cw].value = item;
        buf_[cw].produced.store(true, std::memory_order_release); // 标记slot已写入
        return true;
    }

    // 已支持多生产多消费（MPMC）——每个槽有独立的produced标志，配合原子操作可安全并发
    bool dequeue(T& item) {
        size_t cr, nr;
        do {
            cr = r_.load(std::memory_order_acquire);
            if (cr == w_.load(std::memory_order_acquire)) 
                return false; // 队列空
            // 如果该槽还没有生产好，直接返回空
            if (!buf_[cr].produced.load(std::memory_order_acquire)) {
                return false;
            }
            nr = mask(cr + 1);
        // 参数说明：
        // cr：当前读指针的位置（消费者的下标）
        // nr：下一位置（cr+1，经过掩码，表示准备推进后的读指针）
        // std::memory_order_acq_rel：CAS操作成功时使用的内存序，确保前后对slot内容的依赖顺序
        // std::memory_order_acquire：CAS操作失败时使用的内存序，无需强同步，只需获得最新的r_值
        } while (!r_.compare_exchange_weak(cr, nr, std::memory_order_acq_rel, std::memory_order_acquire));
        item = buf_[cr].value;
        buf_[cr].produced.store(false, std::memory_order_release); // 标记已消费
        return true;
    }

    bool empty() const {
        return r_.load() == w_.load();
    }

    bool full() const {
        return mask(w_.load() + 1) == r_.load();
    }

    size_t size() const {
        size_t w = w_.load();
        size_t r = r_.load();
        if (w >= r) {
            return w - r;
        } else {
            return Size - (r - w);
        }
    }

    size_t capacity() const {
        return Size;
    }

    // 单线程安全的入队操作(简单)
    bool enqueue_single(const T& item) {
        size_t cw = w_;
        size_t nw = mask(cw + 1);
        if (nw == r_) return false;
        buf_[cw].value = item;
        buf_[cw].produced.store(true, std::memory_order_release);
        w_ = nw;
        return true;
    }
};

#include <iostream>
#include <thread>
#include <vector>
#include <atomic>
#include <cassert>

// 假设LockFreeArrayQueue定义在上方

int main() {
    constexpr size_t QUEUE_SIZE = 8;
    LockFreeArrayQueue<int, QUEUE_SIZE> queue;

    // 启动2个生产者和2个消费者线程
    std::atomic<bool> start_flag{false};
    std::vector<int> produced, consumed;
    std::mutex prod_mutex, cons_mutex;

    auto producer = [&](int base) {
        while (!start_flag.load()) std::this_thread::yield();
        for (int i = 0; i < 20; ++i) {
            int val = base + i;
            while (!queue.enqueue(val)) std::this_thread::yield(); // busy-wait
            {
                std::lock_guard<std::mutex> lock(prod_mutex);
                produced.push_back(val);
            }
        }
    };

    auto consumer = [&]() {
        while (!start_flag.load()) std::this_thread::yield();
        int cnt = 0;
        while (cnt < 20) {
            int val;
            if (queue.dequeue(val)) {
                {
                    std::lock_guard<std::mutex> lock(cons_mutex);
                    consumed.push_back(val);
                }
                ++cnt;
            } else {
                std::this_thread::yield();
            }
        }
    };

    std::thread p1(producer, 1000), p2(producer, 2000);
    std::thread c1(consumer), c2(consumer);

    start_flag.store(true);
    p1.join(); p2.join(); c1.join(); c2.join();

    std::cout << "Produced count: " << produced.size() << std::endl;
    std::cout << "Consumed count: " << consumed.size() << std::endl;

    std::sort(produced.begin(), produced.end());
    std::sort(consumed.begin(), consumed.end());
    assert(produced == consumed);
    std::cout << "LockFreeArrayQueue 多线程生产/消费验证通过!" << std::endl;
    return 0;
}
```

// ABA问题示例
// 线程1: 读取head=A，准备CAS
// 线程2: pop A, pop B, push A（A被重新放回）
// 线程1: CAS成功，但是B已经被释放

// 解决方案1: 使用版本号/时间戳
template<typename T>
struct TaggedPointer {
    T* ptr;
    size_t tag;  // 版本号
};

std::atomic<TaggedPointer<Node>> head;

void push(Node* node) {
    TaggedPointer<Node> old_head = head.load();
    TaggedPointer<Node> new_head;
    do {
        node->next = old_head.ptr;
        new_head = {node, old_head.tag + 1};
    } while (!head.compare_exchange_weak(old_head, new_head));

## 双重检查锁定

```cpp
#include <atomic>
#include <mutex>

class Singleton {
private:
    static std::atomic<Singleton*> instance;
    static std::mutex mtx;
    
    Singleton() = default;

public:
    static Singleton* getInstance() {
        Singleton* tmp = instance.load(std::memory_order_acquire);
        if (!tmp) {
            std::lock_guard<std::mutex> lock(mtx);
            tmp = instance.load(std::memory_order_relaxed);
            if (!tmp) {
                tmp = new Singleton();
                instance.store(tmp, std::memory_order_release);
            }
        }
        return tmp;
    }
};

std::atomic<Singleton*> Singleton::instance{nullptr};
std::mutex Singleton::mtx;
```
