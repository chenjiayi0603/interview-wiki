# 网络优化技术

## 4.1 零拷贝技术

**原理**：减少数据在内核态和用户态之间的拷贝次数。

**技术**：
- **sendfile()**：文件到网络直接传输
- **mmap()**：内存映射，避免拷贝
- **splice()**：管道间零拷贝传输

```cpp
// sendfile + mmap 零拷贝完整例子（从本地文件零拷贝发送到 socket）

#include <sys/sendfile.h>
#include <sys/mman.h>
#include <fcntl.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <cstdio>
#include <cstring>
#include <cstdlib>
#include <cerrno>

// 服务端：接收客户端请求，并使用 sendfile 零拷贝发送文件
void run_sendfile_server(const char* filename, int port) {
    int file_fd = open(filename, O_RDONLY);
    if (file_fd < 0) {
        perror("open file");
        exit(1);
    }

    // 获取文件大小
    off_t filesize = lseek(file_fd, 0, SEEK_END);
    lseek(file_fd, 0, SEEK_SET);

    // 创建 TCP socket
    int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (listen_fd < 0) {
        perror("socket");
        exit(1);
    }
    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(port);

    if (bind(listen_fd, (sockaddr*)&addr, sizeof(addr)) < 0) {
        perror("bind");
        exit(1);
    }
    listen(listen_fd, 1);

    printf("Server listening on port %d, waiting for connection...\n", port);
    int client_fd = accept(listen_fd, NULL, NULL);
    if (client_fd < 0) {
        perror("accept");
        exit(1);
    }
    printf("Client connected, sending file...\n");
    // 使用 sendfile 零拷贝传输文件
    // 参数说明：
    // socket_fd  : 目标 socket 文件描述符
    // file_fd    : 源文件的文件描述符（不一定非得是文件，也可以是比如 pipe、memfd_create 创建的内存 fd 等，只要 sendfile 支持，理论上可用于多种类型的数据传递）
    // offset     : 指向文件偏移量的指针，可为 NULL（表示从当前偏移）
    // count      : 要发送的字节数
    off_t offset = 0;
    ssize_t sent = 0;
    while (offset < filesize) {
        sent = sendfile(client_fd, file_fd, &offset, filesize - offset);
        if (sent <= 0) {
            perror("sendfile");
            break;
        }
    }

    printf("Sendfile done. Total sent: %lld bytes\n", (long long)offset);

    close(client_fd);
    close(listen_fd);
    close(file_fd);
}

// 客户端：连接服务器并使用 mmap 零拷贝读出接收到的数据
void run_receive_and_mmap(const char* output_file, const char* server_ip, int port) {
    int sock = socket(AF_INET, SOCK_STREAM, 0);
    sockaddr_in server_addr{};
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);
    inet_pton(AF_INET, server_ip, &server_addr.sin_addr);

    if (connect(sock, (sockaddr*)&server_addr, sizeof(server_addr)) < 0) {
        perror("connect");
        exit(1);
    }

    // 接收数据写到 output_file
    int out_fd = open(output_file, O_RDWR | O_CREAT | O_TRUNC, 0666);
    if (out_fd < 0) {
        perror("open output");
        exit(1);
    }

    char buffer[4096];
    ssize_t n;
    size_t total = 0;
    // 客户端这里虽然后续用到了 mmap 做文件读取, 但网络接收部分 (recv) 实际并非零拷贝——数据是先从内核缓冲区拷贝到用户态 buffer，再写入文件（write 时可能再次拷贝）。
    while ((n = recv(sock, buffer, sizeof(buffer), 0)) > 0) {
        write(out_fd, buffer, n);
        total += n;
    }
    printf("Received %zu bytes\n", total);

    // mmap 读取刚刚保存的文件
    // 数据已经写入 output_file，此处通过 mmap 将文件映射到内存，避免 read 再次将文件内容拷贝到用户缓冲区，实现高效文件访问（类似只读“零拷贝”处理本地大文件场景）

    // mmap本身是"本地文件到用户态"的零拷贝技术（读取本地文件内容时可避免read系统调用和用户空间缓冲区拷贝）
    // 但此例的网络接收部分仍有数据从内核拷贝到用户空间（recv和write是非零拷贝）；mmap只针对已经落盘的文件映射到内存时实现零拷贝
    void* mapped = mmap(NULL, total, PROT_READ, MAP_PRIVATE, out_fd, 0);  // mmap实现了文件内容“就地”映射，读文件时无需额外拷贝
    if (mapped == MAP_FAILED) {
        perror("mmap");
        exit(1);
    }

    // 假设是文本文件，打印前 100 字节内容
    printf("First 100 bytes from mmap:\n");
    fwrite(mapped, 1, total < 100 ? total : 100, stdout);
    printf("\n");

    munmap(mapped, total);
    close(out_fd);
    close(sock);
}

// 示例用法：
//   终端1： ./a.out server file.txt 12345
//   终端2： ./a.out client recv.txt 127.0.0.1 12345
int main(int argc, char* argv[]) {
    if (argc >= 4 && strcmp(argv[1], "server") == 0) {
        const char* filename = argv[2];
        int port = atoi(argv[3]);
        run_sendfile_server(filename, port);
    } else if (argc >= 5 && strcmp(argv[1], "client") == 0) {
        const char* output = argv[2];
        const char* server_ip = argv[3];
        int port = atoi(argv[4]);
        run_receive_and_mmap(output, server_ip, port);
    } else {
        printf("用法:\n");
        printf("  %s server <file> <port>\n", argv[0]);
        printf("  %s client <savefile> <server_ip> <port>\n", argv[0]);
    }
    return 0;
}
```

## 4.2 用户态网络栈

**原理**：绕过内核网络栈，直接在用户态处理网络数据包。

**技术栈**：
- **DPDK（Data Plane Development Kit）**：Intel 开发的用户态网络框架
- **PF_RING**：高性能数据包捕获
- **XDP（eXpress Data Path）**：Linux 内核 eBPF 技术

**DPDK 示例**：
DPDK（Data Plane Development Kit）是 Intel 为了解决传统内核网络协议栈处理性能瓶颈而设计的高性能用户态网络开发框架。其核心目的在于：

- **绕过内核协议栈**，直接在用户空间收发网络数据包，极大降低系统调用和中断带来的延迟。
- **最大化吞吐与最低延迟**：使单个普通服务器即可实现 10G/40G/100G 网络接口线速数据包处理，延迟降至微秒甚至纳秒级。
- **满足高频交易、网络设备、转发网关等对极低时延/高并发有极致需求的场景。**

主要实现方式包括：
- 将网卡 DMA 数据缓冲区直接映射给用户态进程，用户进程可直接操作网络收发队列，无需频繁内核切换。
- 提供高效的内存池、无锁队列和CPU亲和优化，减少资源竞争和缓存失效。
- 完全跳过 Linux 内核网络协议栈，避免内核状态切换开销。

**总结一句话**：DPDK 让你的网络数据处理“只在用户态、零拷贝、极快极稳”——它就是为了追求极致吞吐性能与极低转发延迟而生。

```cpp
// DPDK 基本流程
// DPDK 完整最小示例（1 收 1 发）
// 编译需链接：-ldpdk，并配置 hugepage、绑定网卡等
#include <stdio.h>
#include <stdint.h>
#include <stdlib.h>
#include <rte_eal.h>
#include <rte_ethdev.h>
#include <rte_mempool.h>
#include <rte_mbuf.h>

#define RX_RING_SIZE 128
#define TX_RING_SIZE 512
#define NUM_MBUFS ((4096 - 1) * 2)
#define MBUF_CACHE_SIZE 250
#define BURST_SIZE 32

static const struct rte_eth_conf port_conf_default = {
    .rxmode = { .max_rx_pkt_len = RTE_ETHER_MAX_LEN }
};

int main(int argc, char* argv[]) {
    int ret = rte_eal_init(argc, argv);
    if (ret < 0) {
        rte_exit(EXIT_FAILURE, "错误: EAL 初始化失败\n");
    }
    argc -= ret;
    argv += ret;

    uint16_t port_id = 0;
    uint16_t nb_ports = rte_eth_dev_count_avail();
    if (nb_ports == 0) {
        rte_exit(EXIT_FAILURE, "错误: 无可用网卡端口\n");
    }

    // 创建 mbuf 内存池
    struct rte_mempool* mbuf_pool = rte_pktmbuf_pool_create("MBUF_POOL",
                                NUM_MBUFS, MBUF_CACHE_SIZE, 0,
                                RTE_MBUF_DEFAULT_BUF_SIZE, rte_socket_id());
    if (mbuf_pool == NULL)
        rte_exit(EXIT_FAILURE, "创建 mbuf pool 失败\n");

    // 配置端口
    struct rte_eth_conf port_conf = port_conf_default;
    if (rte_eth_dev_configure(port_id, 1, 1, &port_conf) < 0)
        rte_exit(EXIT_FAILURE, "网卡配置失败\n");

    // 配置 RX 队列
    if (rte_eth_rx_queue_setup(port_id, 0, RX_RING_SIZE,
        rte_eth_dev_socket_id(port_id), NULL, mbuf_pool) < 0)
        rte_exit(EXIT_FAILURE, "RX 队列配置失败\n");

    // 配置 TX 队列
    if (rte_eth_tx_queue_setup(port_id, 0, TX_RING_SIZE,
        rte_eth_dev_socket_id(port_id), NULL) < 0)
        rte_exit(EXIT_FAILURE, "TX 队列配置失败\n");

    // 启动端口
    if (rte_eth_dev_start(port_id) < 0)
        rte_exit(EXIT_FAILURE, "启动网卡失败\n");

    printf("端口 %u 已启动, 等待收发包...\n", port_id);

    // 主循环：收包后直接发回
    while (1) {
        struct rte_mbuf* bufs[BURST_SIZE];
        uint16_t nb_rx = rte_eth_rx_burst(port_id, 0, bufs, BURST_SIZE);
        if (nb_rx > 0) {
            // 这里可以对 bufs 里的包做处理
            uint16_t nb_tx = rte_eth_tx_burst(port_id, 0, bufs, nb_rx);
            // 释放未发送成功的 mbufs
            if (nb_tx < nb_rx) {
                for (uint16_t i = nb_tx; i < nb_rx; i++)
                    rte_pktmbuf_free(bufs[i]);
            }
        }
    }

    // 程序永远不会到这里，实际需加信号处理安全退出
    // rte_eth_dev_stop(port_id);
    // rte_eth_dev_close(port_id);
    // rte_mempool_free(mbuf_pool);
    // rte_eal_cleanup();
    return 0;
}
```

```cpp
//   - 如果DPDK服务器监听的是虚拟设备（如 net_ring0），则客户端也应用DPDK、同样用--vdev=net_ring0参数启动，两端就像“同一块网卡”通信。

#include <stdint.h>
#include <stdlib.h>
#include <rte_eal.h>
#include <rte_ethdev.h>
#include <rte_mempool.h>
#include <rte_mbuf.h>
#include <rte_ether.h>

#define RX_RING_SIZE 128
#define TX_RING_SIZE 512
#define NUM_MBUFS ((4096 - 1) * 2)
#define MBUF_CACHE_SIZE 250
#define BURST_SIZE 32

static const struct rte_eth_conf port_conf_default = {
    .rxmode = { .max_rx_pkt_len = RTE_ETHER_MAX_LEN }
};

int main(int argc, char* argv[]) {
    if (argc < 3) {
        printf("用法: %s <port_id> <send_count> [EAL参数...]\n", argv[0]);
        printf("\n");
        printf("重要: 如果服务器(test1)正在运行，必须使用 --file-prefix 参数！\n");
        printf("例如（虚拟网卡模式）:\n");
        printf("  sudo %s 0 100 --vdev=net_ring0 --no-pci --lcores=0 --file-prefix=client\n", argv[0]);
        printf("\n");
        printf("如果遇到 'Cannot create lock' 错误:\n");
        printf("  1. 使用 --file-prefix=client 参数（推荐）\n");
        printf("  2. 或停止其他DPDK进程: sudo pkill -f test1\n");
        return 1;
    }
    // 解析参数
    uint16_t port_id = (uint16_t)atoi(argv[1]);
    int send_count = atoi(argv[2]);

    // DPDK EAL 初始化
    // 注意: 如果另一个DPDK进程正在运行，需要添加 --file-prefix 参数避免冲突
    // 例如: --file-prefix=client
    int ret = rte_eal_init(argc, argv);
    if (ret < 0) {
        rte_exit(EXIT_FAILURE, "EAL 初始化失败\n");
    }
    argc -= ret;
    argv += ret;

    // 检查端口
    if (port_id >= rte_eth_dev_count_avail()) {
        rte_exit(EXIT_FAILURE, "无效的port_id\n");
    }

    // 创建 mbuf pool（用于发送和接收）
    struct rte_mempool* mbuf_pool = rte_pktmbuf_pool_create("MBUF_POOL",
                                NUM_MBUFS, MBUF_CACHE_SIZE, 0,
                                RTE_MBUF_DEFAULT_BUF_SIZE, rte_socket_id());
    if (mbuf_pool == NULL)
        rte_exit(EXIT_FAILURE, "mbuf pool 创建失败\n");

    // 配置端口（需要至少1个RX队列和1个TX队列）
    if (rte_eth_dev_configure(port_id, 1, 1, &port_conf_default) < 0)
        rte_exit(EXIT_FAILURE, "网卡配置失败\n");

    // 配置 RX 队列（即使不接收，也需要配置以确保端口正常工作）
    if (rte_eth_rx_queue_setup(port_id, 0, RX_RING_SIZE,

    // 配置发送队列
    if (rte_eth_tx_queue_setup(port_id, 0, TX_RING_SIZE, rte_eth_dev_socket_id(port_id), NULL) < 0)

    struct rte_ether_addr src_mac, dst_mac;
    if (rte_eth_macaddr_get(port_id, &src_mac) != 0) {

    printf("客户端端口 %u MAC地址: %02X:%02X:%02X:%02X:%02X:%02X\n", port_id,

    printf("目标MAC地址设置为广播地址: FF:FF:FF:FF:FF:FF\n");
    fflush(stdout);

    // 主循环：发送数据包
    for (int i = 0; i < send_count; ++i) {
        // 创建mbuf
        struct rte_mbuf* mbuf = rte_pktmbuf_alloc(mbuf_pool);
        if (!mbuf) {
            printf("mbuf分配失败\n");
            continue;
        }

        // 构造以太网头部
        struct rte_ether_hdr* eth_hdr = (struct rte_ether_hdr*)rte_pktmbuf_append(mbuf, sizeof(struct rte_ether_hdr));
        if (!eth_hdr) {
            printf("无法添加以太网头部\n");
            rte_pktmbuf_free(mbuf);
            continue;
        }
        
        // 设置以太网头部字段
        rte_ether_addr_copy(&dst_mac, &eth_hdr->d_addr);
        rte_ether_addr_copy(&src_mac, &eth_hdr->s_addr);
        eth_hdr->ether_type = rte_cpu_to_be_16(RTE_ETHER_TYPE_IPV4); // 使用IPv4类型，实际可以是任意值

        // 填充测试数据（在以太网头部之后）
        const char* msg = "Hello, this is a low latency DPDK test message.";
        uint16_t msg_len = (uint16_t)strlen(msg) + 1;
        char* pkt_data = (char*)rte_pktmbuf_append(mbuf, msg_len);
        if (!pkt_data) {
            printf("无法添加数据\n");
            rte_pktmbuf_free(mbuf);
            continue;
        }
        memcpy(pkt_data, msg, msg_len);

        // 发送包
        uint16_t nb_tx = rte_eth_tx_burst(port_id, 0, &mbuf, 1);
        if (nb_tx < 1) {
            printf("发送失败 (第 %d 个包)，可能发送队列已满\n", i + 1);
            rte_pktmbuf_free(mbuf);
            // 如果发送失败，等待一下再重试
            usleep(1000);
        } else {
            printf("已发送第 %d 个数据包 (总长度: %u 字节)\n", i + 1, rte_pktmbuf_pkt_len(mbuf));
            fflush(stdout);
            // 添加小延迟，确保数据包有时间传输
            usleep(100); // 延迟100微秒
        }
    }
    printf("数据包发送完成, send_count=%d\n", send_count);

    rte_eth_dev_stop(port_id);
    rte_eth_dev_close(port_id);
    rte_mempool_free(mbuf_pool);
    rte_eal_cleanup();
    return 0;
}
```

## 4.3 批量处理

**原理**：一次处理多个数据包，减少系统调用和上下文切换开销。

```cpp
// 优化前：逐个处理
void process_packets_naive(int socket_fd) {
    char buffer[1024];
    while (true) {
        ssize_t n = recv(socket_fd, buffer, sizeof(buffer), 0);
        process_packet(buffer, n);
    }
}

// 优化后：批量处理
void process_packets_batch(int socket_fd) {
    char buffers[32][1024];
    struct iovec iovs[32];
    for (int i = 0; i < 32; ++i) {
        iovs[i].iov_base = buffers[i];
        iovs[i].iov_len = 1024;
    }
    
    while (true) {
        // 批量接收（readv）
        ssize_t n = readv(socket_fd, iovs, 32);
        // 批量处理
        for (ssize_t i = 0; i < n; ++i) {
            process_packet(buffers[i], iovs[i].iov_len);
        }
    }
}
```

## 4.4 网络缓冲区优化

**策略**：
- 增大 socket 缓冲区
- 使用自定义缓冲区管理
- 避免数据拷贝

```cpp
#include <sys/socket.h>

// 增大 socket 缓冲区
void optimize_socket_buffers(int socket_fd) {
    int send_buf_size = 1024 * 1024;  // 1MB
    int recv_buf_size = 1024 * 1024;   // 1MB
    
    setsockopt(socket_fd, SOL_SOCKET, SO_SNDBUF, 
               &send_buf_size, sizeof(send_buf_size));
    setsockopt(socket_fd, SOL_SOCKET, SO_RCVBUF, 
               &recv_buf_size, sizeof(recv_buf_size));
    
    // 禁用 Nagle 算法（低延迟场景）
    int flag = 1;
    setsockopt(socket_fd, IPPROTO_TCP, TCP_NODELAY, &flag, sizeof(flag));
```
