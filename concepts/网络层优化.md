```cpp
// 轮询接收数据包（无中断延迟）
void poll_packets(uint16_t port_id) {
    struct rte_mbuf* mbufs[32];
    
    while (true) {
        uint16_t nb_rx = rte_eth_rx_burst(port_id, 0, mbufs, 32);
        
        for (uint16_t i = 0; i < nb_rx; ++i) {
            process_packet(mbufs[i]);
            rte_pktmbuf_free(mbufs[i]);
        }
    }
```

#### DPDK 核心原理

DPDK（Data Plane Development Kit）是 Intel 为了解决传统内核网络协议栈处理性能瓶颈而设计的高性能用户态网络开发框架。其核心目的在于：

- **绕过内核协议栈**，直接在用户空间收发网络数据包，极大降低系统调用和中断带来的延迟。
- **最大化吞吐与最低延迟**：使单个普通服务器即可实现 10G/40G/100G 网络接口线速数据包处理，延迟降至微秒甚至纳秒级。
- **满足高频交易、网络设备、转发网关等对极低时延/高并发有极致需求的场景。**

主要实现方式包括：
- 将网卡 DMA 数据缓冲区直接映射给用户态进程，用户进程可直接操作网络收发队列，无需频繁内核切换。
- 提供高效的内存池、无锁队列和CPU亲和优化，减少资源竞争和缓存失效。
- 完全跳过 Linux 内核网络协议栈，避免内核状态切换开销。

**总结一句话**：DPDK 让你的网络数据处理“只在用户态、零拷贝、极快极稳”——它就是为了追求极致吞吐性能与极低转发延迟而生。

### RDMA (Remote Direct Memory Access)

```cpp
#include <infiniband/verbs.h>
#include <string>
#include <cstring>
#include <cstdio>

/*
RDMA 连接建立不是通过 IP+端口建立的，实际需要 out-of-band（带外）方式先通过以太网（如 TCP Socket）交换底层参数（如 LID、GID、QP号、PSN、rkey/addr等），
然后双方通过 ibv_modify_qp 配置 QP 进入 RTS（Ready to Send）状态，才能进行直接 RDMA 通信。

下面只演示传统 TCP 辅助下参数交换、和 RDMA 连接的必要流程片段。IP地址通常只是参数交换用，RDMA通信本身与IP/端口无关。

完整的 RDMA 连接过程（Verbs RC）：
  // RDMA连接的原理简述：
  // RDMA通信中，双方通过“互相交换底层硬件参数”，直接让网卡/主机的内存区域建立一对一高速映射，不再经过内核协议栈——
  // 能做到“对端主机内存直接读写”，极大降低了延迟。
  //
  // RDMA参数详解：
  // - LID（Local Identifier）：本地InfiniBand子网中每个端口分配的物理地址，相当于“局域网内地址”，用于数据包在Fabric内路由到目标网卡端口。
  // - QPN（Queue Pair Number）：每个QP（队列对，数据收发的逻辑通道）的唯一编号，通信双方要明确对方QPN才能指哪发哪。
  // - PSN（Packet Sequence Number）：数据包的起始序列号，RDMA协议用于保证数据顺序与重传校验，避免丢包/乱序。
  // - rkey/addr：远端主机注册的物理内存key/虚拟地址，用于RDMA直接读写时在硬件级别确保权限正确与访问定位。
  //
  // 连接建立流程：
  // 1. 双方各自创建QP，向HCA注册本地用于收发的数据缓冲区（即注册内存，获得rkey/addr等信息），然后通过TCP等方式互换LID、QPN、PSN等参数。
  // 2. 用ibv_modify_qp将本地QP配置到“RTR/RTS”等状态，使其具备收、发能力（包括链路地址、目的QP等填入），此时RDMA通信链路才建立完成。
  3. 交换 rkey/addr 后，即可写入/读取远端注册内存

这里只展示核心流程，实际使用时双方需额外建立 TCP socket 做参数同步。
*/

// 辅助结构体：交换的 RDMA 连接参数
struct RdmaEndpointInfo {
    uint16_t lid;
    uint32_t qpn;
    uint32_t psn;
    uint32_t rkey;
    uintptr_t vaddr;
};

class RdmaConnection {
    struct ibv_context* ctx_;
    struct ibv_pd* pd_;
    struct ibv_cq* cq_;
    struct ibv_qp* qp_;
    struct ibv_mr* mr_;
    void* buf_;
    size_t buf_size_;
    RdmaEndpointInfo local_info_;
    RdmaEndpointInfo remote_info_;

public:
    RdmaConnection() : ctx_(nullptr), pd_(nullptr), cq_(nullptr), qp_(nullptr),
                       mr_(nullptr), buf_(nullptr), buf_size_(4096) {
        memset(&local_info_, 0, sizeof(local_info_));
        memset(&remote_info_, 0, sizeof(remote_info_));
    }

    // Endpoint 参数包括：LID（本地标识符）、QPN（队列对号）、PSN（包序号）、rkey（远端内存权限键）、vaddr（内存虚拟地址）
    // 初始化本地 RDMA 资源时，会生成上述 Endpoint 参数，用于与对端完成连接与直接内存访问。
    bool init_local() {
        int num_devices = 0;
        ibv_device** dev_list = ibv_get_device_list(&num_devices);
        if (num_devices <= 0) {
            printf("No RDMA device found\n");
            return false;
        }

        ctx_ = ibv_open_device(dev_list[0]);
        ibv_free_device_list(dev_list);

        if (!ctx_) {
            printf("Could not open RDMA device\n");
            return false;
        }

        pd_ = ibv_alloc_pd(ctx_);
        cq_ = ibv_create_cq(ctx_, 16, nullptr, nullptr, 0);

        buf_ = malloc(buf_size_);
        mr_ = ibv_reg_mr(pd_, buf_, buf_size_,
            IBV_ACCESS_LOCAL_WRITE | IBV_ACCESS_REMOTE_WRITE | IBV_ACCESS_REMOTE_READ);
        if (!mr_) {
            printf("Failed to register memory\n");
            return false;
        }

        // 获取保护域所属 port 的 lid
        ibv_port_attr port_attr;
        ibv_query_port(ctx_, 1, &port_attr);
        local_info_.lid = port_attr.lid;

        // 创建 QP
        ibv_qp_init_attr qp_attr = {};
        qp_attr.send_cq = cq_;
        qp_attr.recv_cq = cq_;
        qp_attr.qp_type = IBV_QPT_RC;
        qp_attr.cap.max_send_wr = 16;
        qp_attr.cap.max_recv_wr = 16;
        qp_attr.cap.max_send_sge = 1;
        qp_attr.cap.max_recv_sge = 1;

        qp_ = ibv_create_qp(pd_, &qp_attr);

        local_info_.qpn = qp_->qp_num;
        local_info_.psn = rand() & 0xffffff;
        local_info_.rkey = mr_->rkey;
        local_info_.vaddr = reinterpret_cast<uintptr_t>(buf_);

        return true;
    }

    // 假设通过 TCP/其他带外方式传来远端参数(remote)
    void set_remote_info(const RdmaEndpointInfo& remote) {
        remote_info_ = remote;
    }

    // RDMA通信前必须进行 QP 步进和参数交换
    // QP（三态）：怎么建立 RDMA 连接？——连接流程实现如下
    // QP “发起”与“接收”双方必须都走以下三个状态变迁，并互通参数（一般通过 TCP 或带外交换）
    // 1. QP->INIT（初始化，可本地完成）
    // 2. QP->RTR（Ready to Receive，设置远端 QP/QPN/lid 等参数，必须有 remote_info）
    // 3. QP->RTS（Ready to Send，发送侧 Ready）

    // 具体过程见如下代码，"connect_qp" 即“怎么链接”：
    // RDMA QP（三态）连接流程：INIT -> RTR -> RTS
    /*
    远端地址指的是什么？  
    一般 RDMA 连接双方需要互通的一组“远端参数”包括：
      - 远端 LID（lid）：Infiniband 层的本地标识符。
      - 远端 QPN（qpn）：Queue Pair Number，用于区分对端的QP。
      - 远端 PSN（psn）：Packet Sequence Number。
      - 远端 rkey ：远端注册内存区的 key。
      - 远端 vaddr：远端注册内存区的虚拟地址（即：只要你有 rkey/vaddr，就可以直接 RDMA 访问对方内存）。

    本例 remote_info_ 结构中含都有。  
    对于“远端地址”，通常指远端注册内存区的虚拟地址（vaddr），比如要 RDMA Write/Read 时，指定 remote_info_.vaddr 和 remote_info_.rkey：

    // 代码中的赋值片段（见init_local）：
    local_info_.vaddr = reinterpret_cast<uintptr_t>(buf_);
    // ...
    local_info_.rkey = mr_->rkey;

    // 远端通过 set_remote_info() 收到远端 info 后，remote_info_.vaddr 就是对方的“远端地址”

    // connect_qp()本身只是进行 QP 的三态互通和链路建立，并没有直接使用远端地址（vaddr）。
    // 真正做 RDMA 写/读时，请参考如下方式：
    // --- RDMA Write/Read 时 ---
    // sge.addr = 本地 buffer 虚拟地址（要发出的数据结构）
    // wr.remote_addr = remote_info_.vaddr;
    // wr.rkey = remote_info_.rkey;
    */

    bool connect_qp() {
        // 1. QP 进入 INIT 状态，设置本地读写权限
        ibv_qp_attr attr = {};
        attr.qp_state = IBV_QPS_INIT;
        attr.pkey_index = 0;                // 分区 key 索引，恒为 0
        attr.port_num = 1;                  // RDMA 端口号
        attr.qp_access_flags = IBV_ACCESS_REMOTE_WRITE | IBV_ACCESS_REMOTE_READ; // 允许远端直接读写
        if (ibv_modify_qp(qp_, &attr,
                IBV_QP_STATE | IBV_QP_PKEY_INDEX | IBV_QP_PORT | IBV_QP_ACCESS_FLAGS)) {
            printf("Failed QP->INIT\n");
            return false;
        }

        // 2. QP 进入 RTR (Ready to Receive) 状态，这一步同样没有传远端内存地址，而只需配置与远端通信有关的参数（如 QPN、PSN、LID 等，不包含远端 vaddr）。
         // 那要是双方 QP 不在同一台服务器，甚至在不同虚拟机（VM）上，这还能连吗？答案
        // 是：可以！RDMA 的连接建立过程（QP 三态转换）只是协议本身要求双方互通标识参数
        // （QPN、PSN、LID、MTU 等），只要网络可达，而且这些参数完成了交换配置，无论在物理
        // 机还是云上的虚拟机、容器，只要 RDMA 网络打通，都可以按此流程建立 QP 连接，实现 
        // RDMA SEND/RECV 消息收发。  


        // QPN（Queue Pair Number）、PSN（Packet Sequence Number）、LID（Local Identifier）、MTU（最大传输单元）与以太网物理地址（MAC地址）其实属于不同的协议层。QPN/PSN/LID/MTU 是 RDMA（如 InfiniBand 或 RoCE）协议栈内部控制和识别通信双方的参数，和以太网物理层的 MAC 地址是正交的。
        // 具体来说：
        // - QPN 是由 RDMA 设备驱动或硬件分配的，每创建一个 QP（Queue Pair）都会分配唯一的 QPN 进行会话标识。
        // - PSN 是初始化连接时，由通信双方随机生成并协商的，用于数据包顺序控制和丢包重传判定。
        // - LID 是 InfiniBand 网络中的本地唯一标识，由子网管理器（Subnet Manager）为每个端口分配，对应一台主机上的一个 HCA 端口。如果是 RoCE（以太网 RDMA），会用 GID/IPv4/IPv6代替LID。
        // - MTU 是根据物理链路实际支持的最大包长协商的，受网络硬件和配置（如交换机支持的 MTU、以太网配置的 jumbo frame）影响，连接双方通过查询本地端口属性或子网管理器进行协商选取一个共同支持的最小值。
        // 以太网的物理地址（MAC）在 RDMA over Ethernet（如 RoCE）场景下用于底层以太网包的转发，与 QPN、PSN、LID/IPv6/GID 没有直接映射关系；RDMA 协议栈会通过 MAC 地址进行数据帧的物理寻址，而 RDMA 通信层再通过 QPN/LID/PSN/MTU 做会话和包管理。
        // 因此：RDMA QP 连接参数（QPN、PSN、LID、MTU）通过 RDMA 协议内部机制协商和分配，不直接用于确定以太网的物理地址，实际物理寻址依赖于底层以太网（RoCE）或 InfiniBand 网络的转发机制。
        // 实际对远端内存的直接访问，只在 RDMA WRITE/READ 时才需知道对方的 vaddr 和 rkey，这与 QP 进入 RTR 和连接建立是正交的两个流程。  
        // 所以不管双方物理隔离如何，只要 RDMA 路由、连通性和上述协议参数无误，都可在不同虚拟机间顺利建立 RDMA QP 连接。
        memset(&attr, 0, sizeof(attr));
        attr.qp_state = IBV_QPS_RTR;
        attr.path_mtu = IBV_MTU_1024;           // 连接 MTU，通常为 1024
        attr.dest_qp_num = remote_info_.qpn;    // 远端 QPN（队列号）
        attr.rq_psn = remote_info_.psn;         // 远端 PSN
        attr.max_dest_rd_atomic = 1;            // 允许最大未完成原子操作数（典型为1）
        attr.min_rnr_timer = 12;                // RNR NAK 最小重试间隔
        attr.ah_attr.is_global = 0;             // 非全局路由
        attr.ah_attr.dlid = remote_info_.lid;   // 远端 LID
        attr.ah_attr.sl = 0;                    // 服务等级（Service Level）
        attr.ah_attr.src_path_bits = 0;
        attr.ah_attr.port_num = 1;              // 本地端口

        if (ibv_modify_qp(qp_, &attr,
                IBV_QP_STATE | IBV_QP_AV | IBV_QP_PATH_MTU | IBV_QP_DEST_QPN | 
                IBV_QP_RQ_PSN | IBV_QP_MAX_DEST_RD_ATOMIC | IBV_QP_MIN_RNR_TIMER)) {
            printf("Failed QP->RTR\n");
            return false;
        }

        // 3. QP 进入 RTS (Ready to Send) 状态，配置发送参数
        memset(&attr, 0, sizeof(attr));
        attr.qp_state = IBV_QPS_RTS;
        attr.timeout = 14;                  // 超时时间
        attr.retry_cnt = 7;                 // 重传次数
        attr.rnr_retry = 7;                 // RNR 重试
        attr.sq_psn = local_info_.psn;      // 本地 PSN
        attr.max_rd_atomic = 1;             // 允许本端最大未完成原子操作数

        if (ibv_modify_qp(qp_, &attr,
                IBV_QP_STATE | IBV_QP_TIMEOUT | IBV_QP_RETRY_CNT |
                IBV_QP_RNR_RETRY | IBV_QP_SQ_PSN | IBV_QP_MAX_QP_RD_ATOMIC)) {
            printf("Failed QP->RTS\n");
            return false;
        }
        return true;
    }

    // 简单演示 RDMA send (仅在双方参数交换、QP完成连接后有效)
    // 对方地址在 send() 方法中并没有直接指定，RDMA send 操作是走 QP 连接语义：只需要双方已建立连接即可根据预先协商的远端数据结构（如 remote_info_.vaddr）做写入。
    // 若需要 RDMA WRITE(写到远端指定内存地址)，则需指定远端vaddr和rkey。
    // send() 的 "对方地址" 实际上是由远端主动 post_recv 接收缓冲区决定，而不是本地显式指定。
    // 如果是 RDMA WRITE/READ，代码如下所示：

    bool write_remote(const void* data, size_t len) {
        if (!qp_ || !mr_ || len > buf_size_) return false;

        memcpy(buf_, data, len);

        struct ibv_sge sge;
        sge.addr = reinterpret_cast<uintptr_t>(buf_);
        sge.length = len;
        sge.lkey = mr_->lkey;

        struct ibv_send_wr wr = {};
        wr.opcode = IBV_WR_RDMA_WRITE;
        wr.sg_list = &sge;
        wr.num_sge = 1;
        // 指定对方虚拟地址（通过参数交换获得）
        wr.wr.rdma.remote_addr = remote_info_.vaddr; // 对方地址
        wr.wr.rdma.rkey = remote_info_.rkey;         // 对方 rkey

        struct ibv_send_wr* bad_wr = nullptr;
        int ret = ibv_post_send(qp_, &wr, &bad_wr);
        if (ret) {
            printf("ibv_post_send (WRITE) error: %d\n", ret);
            return false;
        }
        return true;
    }

    // 仍保留原有 send()（用于传统消息，但对方地址不在参数内）
    // send() 负责通过RDMA QP通道以消息语义发送数据到对端，依赖于对方事先用 post_recv 挂接好接收缓冲区，不需要本地显式指定远端地址，常用于传统的消息传递（send/recv）模型
    bool send(const void* data, size_t len) {
        if (!qp_ || !mr_ || len > buf_size_) return false;

        memcpy(buf_, data, len);

        struct ibv_sge sge;
        sge.addr = reinterpret_cast<uintptr_t>(buf_);
        sge.length = len;
        sge.lkey = mr_->lkey;

        struct ibv_send_wr wr = {};
        wr.opcode = IBV_WR_SEND;
        wr.sg_list = &sge;
        wr.num_sge = 1;

        struct ibv_send_wr* bad_wr = nullptr;
        // post_recv 指的是对端主机在 RDMA QP 上预先调用 ibv_post_recv()，将接收缓冲区挂到接收队列。当本端通过 RDMA send 操作发送数据时，数据会自动写入这些已挂载的缓冲区，无需指定远端地址。只有提前完成 post_recv，send 才能成功完成一对一的消息传递。
        // RDMA send 不需要显式指定目标地址，数据写入由对端提前 post_recv 的缓冲区决定
        int ret = ibv_post_send(qp_, &wr, &bad_wr);
        if (ret) {
            printf("ibv_post_send error: %d\n", ret);
            return false;
        }
        return true;
    }

    ~RdmaConnection() {
        if (qp_) ibv_destroy_qp(qp_);
        if (cq_) ibv_destroy_cq(cq_);
        if (mr_) ibv_dereg_mr(mr_);
        if (pd_) ibv_dealloc_pd(pd_);
        if (ctx_) ibv_close_device(ctx_);
        if (buf_) free(buf_);
    }

    const RdmaEndpointInfo& local_info() const { return local_info_; }
    void print_local_info() const {
        printf("[本地 RDMA] lid=%u qpn=%u psn=%u rkey=0x%x vaddr=0x%lx\n",
            local_info_.lid, local_info_.qpn, local_info_.psn,
            local_info_.rkey, local_info_.vaddr);
    }
};

1. 客户端/服务端各自 new RdmaConnection，调用 init_local()；
2. 通过 TCP 互相发送本地 local_info()，recv 到对方信息，set_remote_info()；

int main() {

    conn0.init_local();
    conn1.init_local();
    conn0.print_local_info();
    conn1.print_local_info();

    if (!conn0.send(msg, strlen(msg) + 1)) {
        printf("RDMA 发送失败\n");
        return 1;
    }
    printf("RDMA 发送完成: %s\n", msg);
    return 0;
}
```

```cpp
#include <mtcp_api.h>

void mtcp_init() {

    // 创建 mTCP 上下文
    mctx_t mctx = mtcp_create_context(0);
    
    // 创建 socket
    int sock = mtcp_socket(mctx, AF_INET, SOCK_STREAM, 0);

}
```
