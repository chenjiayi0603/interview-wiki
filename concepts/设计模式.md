## 一、设计原则（SOLID）

### 1. 单一职责原则（SRP）
```cpp
// 违反SRP：一个类做太多事
class User {
public:
    void save() { /* 保存用户 */ }
    void sendEmail() { /* 发送邮件 */ }
    void generateReport() { /* 生成报告 */ }
};

// 遵循SRP：职责分离
class UserRepository { void save(const User&) { } };
class EmailService { void send(const User&) { } };
class ReportGenerator { void generate(const User&) { } };
```

### 2. 开闭原则（OCP）
```cpp
// 违反OCP：扩展需要修改类
class Shape {
public:
    enum Type { CIRCLE, SQUARE };
    Type type;
    double radius;
    double side;
    double area() {
        if (type == CIRCLE) return 3.14 * radius * radius;
        if (type == SQUARE) return side * side;
    }
};

// 遵循OCP：对扩展开放，对修改关闭
class Shape { public: virtual double area() = 0; };

class Circle : public Shape {
    double radius;
public:
    double area() override { return 3.14 * radius * radius; }
};

class Square : public Shape {
    double side;
public:
    double area() override { return side * side; }
};
```

### 3. 里氏替换原则（LSP）
```cpp
// 违反LSP：子类行为不一致
class Rectangle {
protected:
    double width, height;
public:
    void setWidth(double w) { width = w; }
    void setHeight(double h) { height = h; }
};

class Square : public Rectangle {
public:
    void setWidth(double w) override {
        width = height = w;  // 违反LSP
    }
    void setHeight(double h) override {
        width = height = h;
    }
};

// 遵循LSP：正方形不应该继承长方形
// 可以继承Shape基类，或用组合
```

### 5. 依赖倒置原则（DIP）
```cpp
// 违反DIP：高层依赖低层
class MySQLDatabase {
public:
    void connect() { }
    void query(const char* sql) { }
};

class UserService {
    MySQLDatabase db;  // 直接依赖

public:
    void connect() override { }

class UserService {
    IDatabase* db;  // 依赖抽象

        db->query("SELECT...");
```

### 1. 单例模式
```cpp
// 线程安全单例（懒汉）
class Singleton {
private:
    static std::atomic<Singleton*> instance;
    static std::mutex mtx;
    Singleton() = default;
public:
    Singleton(const Singleton&) = delete;
    Singleton& operator=(const Singleton&) = delete;
    
    static Singleton* getInstance() {
        std::lock_guard<std::mutex> lock(mtx);
        if (instance == nullptr) {
            instance = new Singleton();
        }
        return instance;
    }
};

// C++11 局部静态变量（推荐）
class Singleton {
private:
    Singleton() = default;
public:
    static Singleton& getInstance() {
        static Singleton instance;
        return instance;
    }
};
```

```cpp
class Observer {
public:
    virtual void update(const std::string& msg) = 0;
};

class Subject {
    std::vector<Observer*> observers;
public:
    void attach(Observer* o) { observers.push_back(o); }
    void detach(Observer* o) {
        observers.erase(std::remove(observers.begin(), observers.end(), o), observers.end());

    void notify(const std::string& msg) {
        for (auto* o : observers) {
            o->update(msg);
        }
    }
};

public:
    ConcreteObserver(const std::string& n) : name(n) {}

    }
};
```

## 五、面试高频问题

### Q1: 单例模式的线程安全实现？
**参考答案**：
- 饿汉：类加载时创建，线程安全
- 懒汉（双检锁）：`volatile`/原子 + 互斥，避免指令重排与重复创建
- C++11 局部静态（Meyers）：标准保证线程安全，日常最省事

```cpp
// 饿汉：进程内全局唯一，启动即构造
struct SingletonEager {
    static SingletonEager& inst() {
        static SingletonEager s;
        return s;
    }
};

// 懒汉双检锁（示意：atomic + mutex 防重排与重复 new）
struct SingletonLazy {
    static SingletonLazy* inst();
};
std::mutex g_mtx;
std::atomic<SingletonLazy*> g_p{nullptr};
SingletonLazy* SingletonLazy::inst() {
    auto* p = g_p.load(std::memory_order_acquire);
    if (!p) {
        std::lock_guard<std::mutex> lk(g_mtx);
        p = g_p.load(std::memory_order_relaxed);
        if (!p) {
            p = new SingletonLazy();
            g_p.store(p, std::memory_order_release);
        }
    }
    return p;
}

// C++11 局部静态（Meyers，推荐）
struct Singleton {
    static Singleton& inst() {
        static Singleton s;
        return s;
    }
};
```

### Q2: 工厂模式和抽象工厂区别？
**参考答案**：
- 工厂方法：一个创建接口 → **一种**产品类型，子类决定具体类
- 抽象工厂：一个工厂接口 → **一族**配套产品（产品族），换工厂换整套风格
- 记忆：产品**等级结构**（继承树）× 产品**族**（UI 主题 A/B）

```cpp
// 工厂方法：只产一种 Button
struct Button {};
struct WinButton : Button {};
struct ButtonFactory {
    virtual std::unique_ptr<Button> create() = 0;
};
struct WinButtonFactory : ButtonFactory {
    std::unique_ptr<Button> create() override { return std::make_unique<WinButton>(); }
};

// 抽象工厂：一次产一族控件
struct Checkbox {};
struct WinCheckbox : Checkbox {};
struct GuiFactory {
    virtual std::unique_ptr<Button> createButton() = 0;
    virtual std::unique_ptr<Checkbox> createCheckbox() = 0;
};
struct WinGuiFactory : GuiFactory {
    std::unique_ptr<Button> createButton() override { return std::make_unique<WinButton>(); }
    std::unique_ptr<Checkbox> createCheckbox() override { return std::make_unique<WinCheckbox>(); }
};
```

### Q3: 装饰器vs继承？
**参考答案**：
- 装饰器：运行时**嵌套包装**，按需叠加能力；继承：编译期固定超类
- 装饰器减轻「每种组合一个子类」的类爆炸；职责拆成多个小装饰器

```cpp
struct VisualComponent { virtual void draw() = 0; };
struct Text : VisualComponent { void draw() override { /* 纯文字 */ } };

struct Decorator : VisualComponent {
    VisualComponent& inner;
    explicit Decorator(VisualComponent& i) : inner(i) {}
};
struct BorderDec : Decorator {
    using Decorator::Decorator;
    void draw() override { /* 画边框 */ inner.draw(); }
};
struct ScrollDec : Decorator {
    using Decorator::Decorator;
    void draw() override { /* 滚动条 */ inner.draw(); }
};
// 运行时组合：BorderDec(ScrollDec(text))，继承很难这样任意排序叠加
```

### Q4: 代理vs装饰器？
**参考答案**：
- 代理：**控制访问**（延迟加载、权限、远程、缓存、计数），对外尽量**像本体**
- 装饰器：**增强行为**（加功能），调用方知道包了一层也无妨
- 结构都像包装，意图不同：代理「替你挡事」，装饰器「替你加料」

```cpp
struct Image { virtual void display() = 0; };

// 代理：真正的大图延迟加载
struct ProxyImage : Image {
    std::unique_ptr<Image> real;
    std::string file;
    void display() override {
        if (!real) real = loadFromDisk(file);  // 访问时才加载
        real->display();
    }
};

// 装饰器：每次 display 额外记日志（改变/增强行为）
struct LoggingImage : Image {
    Image& inner;
    void display() override { log("show"); inner.display(); }
};
```

### Q5: 观察者vs发布订阅？
**参考答案**：
- 观察者：`Subject` **直接持有** `Observer` 列表，一对多推送
- 发布订阅：发布者、订阅者 **不互相认识**，中间 **Broker/事件总线** 按主题分发，更松耦合、可跨进程

```cpp
// 观察者：主题知道观察者
struct Observer { virtual void onEvent(int x) = 0; };
struct Subject {
    std::vector<Observer*> obs;
    void add(Observer* o) { obs.push_back(o); }
    void notify(int x) { for (auto* o : obs) o->onEvent(x); }
};

// 发布订阅（极简）：中间 map 主题 -> 回调
using Handler = std::function<void(const std::string& payload)>;
std::unordered_map<std::string, std::vector<Handler>> bus;
void publish(const std::string& topic, const std::string& payload) {
    for (auto& h : bus[topic]) h(payload);
}
```

### Q6: 策略vs状态？
**参考答案**：
- 策略：客户端**主动注入**算法，策略间**通常无转移关系**
- 状态：对象内部状态变化 → 行为变化，状态对象常**知道下一状态**（转移规则在内）

```cpp
// 策略：外部随时替换「怎么算」
struct PayContext {
    struct Strategy { virtual int calc(int n) = 0; };
    Strategy* s;
    void set(Strategy* ns) { s = ns; }
    int checkout(int n) { return s->calc(n); }
};

// 状态：内部 request 触发迁移；不同状态类行为不同
struct Turnstile {
    struct State { virtual void coin(Turnstile& m) = 0; };
    State* st;
    void coin() { st->coin(*this); }
};
```

### Q7: 什么是依赖注入？
**参考答案**：

- 依赖注入（DI）：是实现控制反转（IoC）的一种方式，本质是把对象需要的依赖，通过构造/接口/属性**从外部传进来**，对象不用自己 new 或查找依赖了。这样利于解耦、单元测试和灵活扩展。  
- 控制反转（IoC）：大原则，指对象依赖的获取、生命周期等交给「外部容器/框架」管理，不再“自己控制”——依赖注入正是IoC常用的具体做法。 让对象的依赖「由外部」管理和注入，不再在类内部直接 new 或依赖具体实现，而是把创建/装配权交给框架或工厂，实现解耦与灵活扩展

```cpp
struct UserRepo { virtual User load(int id) = 0; };
struct UserService {
    UserRepo& repo;  // 不在内部 new SqlRepo()
    explicit UserService(UserRepo& r) : repo(r) {}
    User get(int id) { return repo.load(id); }
};
// 测试注入 FakeRepo；生产注入 SqlRepo —— 同一套 UserService
```

### Q8: 如何选择设计模式？
**参考答案**：
1. 先写清**变化点**（需求里哪块最常改、哪块要扩展）
2. 用 **SOLID** 衡量：是否单一职责、是否依赖抽象、开闭是否友好
3. **能函数/组合解决就不上模式**；重复三次再抽象
4. 对照「坏味道」：if-else 链→策略/状态；对象树统一操作→组合；频繁 new 族对象→工厂

```cpp
// 小例：分支爆炸时优先考虑策略，而不是继续堆 if
int price(const Order& o, const std::string& userType) {
    if (userType == "vip") return (int)(o.amount * 0.8);
    if (userType == "svip") return (int)(o.amount * 0.7);
    return o.amount;
}
```
