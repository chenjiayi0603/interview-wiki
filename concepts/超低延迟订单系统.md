## 无锁订单队列

```cpp
class LockFreeOrderQueue {
    static constexpr size_t CAPACITY = 4096;
    static constexpr size_t MASK = CAPACITY - 1;
    
    alignas(64) std::atomic<uint64_t> head_{0};
    alignas(64) std::atomic<uint64_t> tail_{0};
    alignas(64) std::array<Order, CAPACITY> buffer_;
    
public:
    bool push(const Order& order) {
        const uint64_t tail = tail_.load(std::memory_order_relaxed);
        const uint64_t next = (tail + 1) & MASK;
        
        if (next == head_.load(std::memory_order_acquire)) {
            return false;  // 队列满
        }
        
        buffer_[tail] = order;
        tail_.store(next, std::memory_order_release);
        return true;
    }
    
    bool pop(Order& order) {
        const uint64_t head = head_.load(std::memory_order_relaxed);
        
        if (head == tail_.load(std::memory_order_acquire)) {
            return false;  // 队列空
        }
        
        order = buffer_[head];
        head_.store((head + 1) & MASK, std::memory_order_release);
        return true;
    }
};
```

## 订单处理器

```cpp
class OrderProcessor {
    LockFreeOrderQueue inbound_;
    LockFreeOrderQueue outbound_;
    
public:
    void run() {
        // 绑定到隔离的 CPU 核心
        set_thread_affinity(2);
        set_realtime_priority(99);
        
        Order order;
        while (true) {
            // 忙等待新订单
            while (!inbound_.pop(order)) {
                _mm_pause();
            }
            
            // 快速处理
            process(order);
            
            // 发送响应
            while (!outbound_.push(order)) {
                _mm_pause();
            }
        }
    }
    
private:
    void process(Order& order) {
        // 验证订单
        if (UNLIKELY(order.quantity == 0 || order.price == 0)) {
            order.status = 'R';  // Rejected
            return;

        // 风控检查（简化示例）

            return;

        send_to_exchange(order);
    }

};
```
